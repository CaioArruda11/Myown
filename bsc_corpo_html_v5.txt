// ============================================================================
// RANKING DINÂMICO DOS QUATRO CARDS DE FAIXA
// ============================================================================
VAR _rankingBase =
    UNION(
        ROW("Cor", "OURO", "Pontos", COALESCE(_pontosOuro, 0)),
        ROW("Cor", "BRONZE", "Pontos", COALESCE(_pontosBronze, 0)),
        ROW("Cor", "PRATA 1", "Pontos", COALESCE(_pontosPrata1, 0)),
        ROW("Cor", "PRATA 2", "Pontos", COALESCE(_pontosPrata2, 0))
    )
VAR _rankingCards =
    ADDCOLUMNS(
        _rankingBase,
        "Posicao",
            VAR _corAtual = [Cor]
            VAR _pontosAtual = [Pontos]
            RETURN
                1 + COUNTROWS(
                    FILTER(
                        _rankingBase,
                        [Pontos] > _pontosAtual
                            || ([Pontos] = _pontosAtual && [Cor] < _corAtual)
                    )
                )
    )

// ============================================================================
// CALCULO (inalterado) - TABELA DOS 12 INDICADORES
// ============================================================================
VAR _tabIndBase =
    ADDCOLUMNS(
        _indicadores,
        "__ResultadoBruto",
            VAR _ind = [Indicador]
            RETURN
                IF(
                    [AnaliticoAT] = 1,
                    CALCULATE(
                        [BSC Resultado Selecionado],
                        REMOVEFILTERS( 'Seletor Cor' ),
                        REMOVEFILTERS( 'Seletor Indicador BSC' ),
                        FILTER(
                            ALL( dAT[COR_AREA ] ),
                            _corSelecionada = "GERAL"
                                || dAT[COR_AREA ] = _corSelecionada
                        ),
                        TREATAS( { _ind }, 'Seletor Indicador BSC'[Indicador] )
                    ),
                    CALCULATE(
                        [BSC Resultado Selecionado],
                        REMOVEFILTERS( 'Seletor Cor' ),
                        REMOVEFILTERS( 'Seletor Indicador BSC' ),
                        REMOVEFILTERS( dAT[AREA ] ),
                        REMOVEFILTERS( dAT[COR_AREA ] ),
                        TREATAS( { _ind }, 'Seletor Indicador BSC'[Indicador] )
                    )
                ),
        "__Meta",
            VAR _ind = [Indicador]
            RETURN
                CALCULATE(
                    [BSC Meta Selecionada],
                    REMOVEFILTERS( 'Seletor Indicador BSC' ),
                    TREATAS( { _ind }, 'Seletor Indicador BSC'[Indicador] )
                ),
        "__MenorMelhor",
            VAR _ind = [Indicador]
            RETURN
                CALCULATE(
                    [BSC Menor Melhor],
                    REMOVEFILTERS( 'Seletor Indicador BSC' ),
                    TREATAS( { _ind }, 'Seletor Indicador BSC'[Indicador] )
                ),
        "__PtsDisp",
            VAR _ind = [Indicador]
            RETURN
                CALCULATE(
                    [BSC Pontos Disponíveis],
                    REMOVEFILTERS( 'Seletor Indicador BSC' ),
                    TREATAS( { _ind }, 'Seletor Indicador BSC'[Indicador] )
                )
    )

// -- guarda contra NaN / -nan(ind) vindo de divisao por zero na medida base
VAR _tabIndResultado =
    ADDCOLUMNS(
        _tabIndBase,
        "__Resultado",
            VAR _bruto = [__ResultadoBruto]
            VAR _invalido = SEARCH( "n", FORMAT( _bruto, "0.000000", "en-US" ), 1, 0 ) > 0
            RETURN
                IF( _invalido, BLANK(), _bruto )
    )

VAR _tabIndAting =
    ADDCOLUMNS(
        _tabIndResultado,
        "__Ating",
            VAR _res = [__Resultado]
            VAR _meta = [__Meta]
            RETURN
                IF(
                    ISBLANK( _res ) || ISBLANK( _meta ),
                    BLANK(),
                    IF(
                        [Indicador] = "IFI 7 DIAS" && _res = 0,
                        1.30,
                        IF(
                            [__MenorMelhor] = 1,
                            DIVIDE( _meta, _res ),
                            DIVIDE( _res, _meta )
                        )
                    )
                )
    )

VAR _tabInd =
    ADDCOLUMNS(
        _tabIndAting,
        "__PtsReais",
            IF(
                ISBLANK( [__Ating] ),
                BLANK(),
                MIN( [__PtsDisp] * 1.30, [__PtsDisp] * [__Ating] )
            ),
        "__Faixa",
            SWITCH(
                TRUE(),
                ISBLANK( [__Ating] ), "neutral",
                [__Ating] >= 1, "green",
                [__Ating] >= 0.90, "yellow",
                [__Ating] >= 0.80, "orange",
                "red"
            )
    )

VAR _qtdIndicadores = COUNTROWS( _indicadores )

VAR _qtdComDados = _qtdIndicadores - _qtdSemDados
