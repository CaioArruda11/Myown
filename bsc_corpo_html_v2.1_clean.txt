// ============================================================================
// RANKING DINÂMICO DOS QUATRO CARDS DE FAIXA
// ============================================================================
VAR _rankingBase =
    UNION(
        ROW("Cor", "OURO", "Pontos", COALESCE(_pontosOuro, 0)),
        ROW("Cor", "BRONZE", "Pontos", COALESCE(_pontosBronze, 0)),
        ROW("Cor", "PRATA 1", "Pontos", COALESCE(_pontosPrata1, 0)),
        ROW("Cor", "PRATA 2", "Pontos", COALESCE(_pontosPrata2, 0))
    )
VAR _rankingCards =
    ADDCOLUMNS(
        _rankingBase,
        "Posicao",
            VAR _corAtual = [Cor]
            VAR _pontosAtual = [Pontos]
            RETURN
                1 + COUNTROWS(
                    FILTER(
                        _rankingBase,
                        [Pontos] > _pontosAtual
                            || ([Pontos] = _pontosAtual && [Cor] < _corAtual)
                    )
                )
    )

// ============================================================================
// CARDS DE FAIXA
// Escala de cor do atingimento (a mesma da tabela e do ranking):
//   green >= 100% | yellow >= 90% | orange >= 80% | red < 80% | neutral = sem dado
// ============================================================================
VAR _faixaGeral =
    SWITCH(
        TRUE(),
        ISBLANK( _atingGeral ), "neutral",
        _atingGeral >= 1, "green",
        _atingGeral >= 0.90, "yellow",
        _atingGeral >= 0.80, "orange",
        "red"
    )
VAR _cardGeral =
    "<div class='card geral" & IF( _corSelecionada = "GERAL", " active", "" ) & "'" &
        " style='--c:" & _corPrimaria & ";--ct:" & _corPrimaria & "'" &
        " title='GERAL: " & FORMAT( _pontosGeral, "0.00", _loc ) & " pts de " & FORMAT( _totalPontosDisponiveis, "0", _loc ) &
        " disponíveis (" & FORMAT( _atingGeral, "0.0%", _loc ) & "). O traço na barra marca a meta de 100%.'>" &
        "<div class='card-top'><i></i><span class='card-name'>GERAL</span><span class='card-sub'>" & FORMAT( _qtdATGeral, "0", _loc ) & " ATs</span></div>" &
        "<div class='card-score'>" & FORMAT( _pontosGeral, "0.00", _loc ) & "<small>pts</small></div>" &
        "<div class='card-meta'><span class='card-cap'>de " & FORMAT( _totalPontosDisponiveis, "0", _loc ) & " disponíveis</span>" &
            "<span class='pill " & _faixaGeral & "'>" & IF( _atingGeral >= 1, "▲ ", "▼ " ) & FORMAT( _atingGeral, "0.0%", _loc ) & "</span></div>" &
        "<div class='track'><u style='width:" & FORMAT( MIN( COALESCE( _atingGeral, 0 ), _tetoBarra ) / _tetoBarra * 100, "0.0", "en-US" ) & "%'></u></div>" &
    "</div>"

// Um unico molde para os 4 cards de faixa, na ordem do ranking dinamico.
VAR _cardsCoresOrdenados =
    CONCATENATEX(
        _rankingCards,
        VAR _cor = [Cor]
        VAR _pts = [Pontos]
        VAR _pos = [Posicao]
        VAR _hex =
            SWITCH( _cor, "OURO", _hexOuro, "BRONZE", _hexBronze, "PRATA 1", _hexPrata1, "PRATA 2", _hexPrata2, _corPrimaria )
        VAR _hexTxt =
            SWITCH( _cor, "OURO", _txtOuro, "BRONZE", _txtBronze, "PRATA 1", _txtPrata1, "PRATA 2", _txtPrata2, _corPrimaria )
        VAR _qtd =
            SWITCH( _cor, "OURO", _qtdATOuro, "BRONZE", _qtdATBronze, "PRATA 1", _qtdATPrata1, "PRATA 2", _qtdATPrata2 )
        VAR _ating = DIVIDE( _pts, _totalPontosDisponiveis )
        VAR _faixa =
            SWITCH(
                TRUE(),
                ISBLANK( _ating ), "neutral",
                _ating >= 1, "green",
                _ating >= 0.90, "yellow",
                _ating >= 0.80, "orange",
                "red"
            )
        VAR _delta = _pts - COALESCE( _pontosGeral, 0 )
        RETURN
            "<div class='card" & IF( _corSelecionada = _cor, " active", "" ) & "'" &
                " style='--c:" & _hex & ";--ct:" & _hexTxt & "'" &
                " title='" & _cor & ": " & FORMAT( _pos, "0", _loc ) & "º lugar entre as faixas · " & FORMAT( _pts, "0.00", _loc ) &
                " pts (" & FORMAT( _ating, "0.0%", _loc ) & ") · " & FORMAT( _delta, "+0.00;-0.00;0.00", _loc ) & " pts em relação ao Geral'>" &
                "<div class='card-top'><i></i><span class='card-name'>" & _cor & "</span>" &
                    "<span class='card-sub'>" & FORMAT( _qtd, "0", _loc ) & IF( _qtd = 1, " AT", " ATs" ) & "</span>" &
                    "<b class='rank'>" & FORMAT( _pos, "0", _loc ) & "º</b></div>" &
                "<div class='card-score'>" & FORMAT( _pts, "0.00", _loc ) & "<small>pts</small></div>" &
                "<div class='card-meta'>" &
                    "<span class='delta " & IF( _delta >= 0, "up", "down" ) & "'>" & FORMAT( _delta, "+0.00;-0.00;0.00", _loc ) & "<small> vs geral</small></span>" &
                    "<span class='pill " & _faixa & "'>" & IF( _ating >= 1, "▲ ", "▼ " ) & FORMAT( _ating, "0.0%", _loc ) & "</span>" &
                "</div>" &
                "<div class='track'><u style='width:" & FORMAT( MIN( COALESCE( _ating, 0 ), _tetoBarra ) / _tetoBarra * 100, "0.0", "en-US" ) & "%'></u></div>" &
            "</div>",
        "",
        [Posicao], ASC
    )

// ============================================================================
// TABELA DOS 12 INDICADORES
// Primeiro monta uma tabela com os valores de cada indicador; depois gera o
// HTML. Assim os subtotais por grupo e os contadores da legenda saem da mesma
// fonte que as linhas exibidas.
// ============================================================================
VAR _tabIndBase =
    ADDCOLUMNS(
        _indicadores,
        "__ResultadoBruto",
            VAR _ind = [Indicador]
            RETURN
                IF(
                    [AnaliticoAT] = 1,
                    CALCULATE(
                        [BSC Resultado Selecionado],
                        REMOVEFILTERS( 'Seletor Cor' ),
                        REMOVEFILTERS( 'Seletor Indicador BSC' ),
                        FILTER(
                            ALL( dAT[COR_AREA ] ),
                            _corSelecionada = "GERAL"
                                || dAT[COR_AREA ] = _corSelecionada
                        ),
                        TREATAS( { _ind }, 'Seletor Indicador BSC'[Indicador] )
                    ),
                    CALCULATE(
                        [BSC Resultado Selecionado],
                        REMOVEFILTERS( 'Seletor Cor' ),
                        REMOVEFILTERS( 'Seletor Indicador BSC' ),
                        REMOVEFILTERS( dAT[AREA ] ),
                        REMOVEFILTERS( dAT[COR_AREA ] ),
                        TREATAS( { _ind }, 'Seletor Indicador BSC'[Indicador] )
                    )
                ),
        "__Meta",
            VAR _ind = [Indicador]
            RETURN
                CALCULATE(
                    [BSC Meta Selecionada],
                    REMOVEFILTERS( 'Seletor Indicador BSC' ),
                    TREATAS( { _ind }, 'Seletor Indicador BSC'[Indicador] )
                ),
        "__MenorMelhor",
            VAR _ind = [Indicador]
            RETURN
                CALCULATE(
                    [BSC Menor Melhor],
                    REMOVEFILTERS( 'Seletor Indicador BSC' ),
                    TREATAS( { _ind }, 'Seletor Indicador BSC'[Indicador] )
                ),
        "__PtsDisp",
            VAR _ind = [Indicador]
            RETURN
                CALCULATE(
                    [BSC Pontos Disponíveis],
                    REMOVEFILTERS( 'Seletor Indicador BSC' ),
                    TREATAS( { _ind }, 'Seletor Indicador BSC'[Indicador] )
                )
    )

// -- guarda contra NaN / -nan(ind) vindo de divisao por zero na medida base
VAR _tabIndResultado =
    ADDCOLUMNS(
        _tabIndBase,

VAR _linhasAreas =
    CONCATENATEX(
        _areasValidas,
        VAR _areaDaLinha = dAT[AREA ]
        VAR _pontosArea = [__Pontos]
        VAR _corArea = [__CorArea]
        // posicao no ranking (mesmo criterio da ordenacao: pontos desc, AT asc)
        VAR _pos =
            1 + COUNTROWS(
                FILTER(
                    _areasValidas,
                    [__Pontos] > _pontosArea
                        || ( [__Pontos] = _pontosArea && dAT[AREA ] < _areaDaLinha )
                )
            )
        VAR _hexCorArea =
            SWITCH(
                _corArea,
                "OURO", _hexOuro,
                "BRONZE", _hexBronze,
                "PRATA 1", _hexPrata1,
                "PRATA 2", _hexPrata2,
                _corPrimaria
            )
        VAR _atingimentoArea = DIVIDE( _pontosArea, _totalPontosDisponiveis )
        VAR _classeArea =
            SWITCH(
                TRUE(),
                ISBLANK( _atingimentoArea ), "neutral",
                _atingimentoArea >= 1, "green",
                _atingimentoArea >= 0.90, "yellow",
                _atingimentoArea >= 0.80, "orange",
                "red"
            )
        VAR _largBarra = FORMAT( DIVIDE( _pontosArea, _maxPtsArea ) * 100, "0.0", "en-US" )
        VAR _selecionada = _temAreaSelecionada && _areaDaLinha = _areaSelecionada
        RETURN
            "<div class='area-row" & IF( _selecionada, " selected", "" ) & IF( _pos <= 3, " top", "" ) & "'" &
                " style='--c:" & _hexCorArea & "'" &
                " title='" & FORMAT( _pos, "0", _loc ) & "º · AT " & _areaDaLinha & " (" & _corArea & "): " &
                FORMAT( _pontosArea, "0.00", _loc ) & " de " & FORMAT( _totalPontosDisponiveis, "0", _loc ) & " pts (" &
                FORMAT( _atingimentoArea, "0.0%", _loc ) & ")'>" &
                "<span class='area-rank'>" & FORMAT( _pos, "0", _loc ) & "</span>" &
                "<div class='area-id'><b>" & _areaDaLinha & "</b><small>" & _corArea & "</small></div>" &
                "<div class='area-pts'>" & FORMAT( _pontosArea, "0.00", _loc ) & "<small>pts</small></div>" &
                "<span class='pill " & _classeArea & "'>" & IF( _atingimentoArea >= 1, "▲ ", "▼ " ) & FORMAT( _atingimentoArea, "0.0%", _loc ) & "</span>" &

            "</div>",
        "",
        [__Pontos], DESC,
        dAT[AREA ], ASC
    )
