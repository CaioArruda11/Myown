// ============================================================================
// BSC - CORPO  (v4)   |   visual 1 de 3: cards de faixa + indicadores + ranking
// Colocar em um HTML Content ocupando o corpo da pagina. Fundo TRANSPARENTE.
// Slicers e botoes continuam nativos do Power BI (fora deste visual).
//
// REGRAS DE CALCULO: identicas as da v2.1. Os blocos marcados "CALCULO" abaixo
// foram copiados sem nenhuma alteracao; so a camada de apresentacao foi
// reescrita.
//
// O QUE MUDOU NA APRESENTACAO
//   - os 12 indicadores cabem na tela sem rolagem: as linhas se ajustam a
//     altura do visual (so rola se o visual ficar pequeno demais);
//   - card GERAL vira o destaque: anel de atingimento (anel fechado = meta) e
//     quanto falta para a meta;
//   - coluna META mostra o sentido do indicador: >= (maior e melhor) ou
//     <= (menor e melhor), sem precisar de etiqueta;
//   - cabecalho da tabela aponta a maior perda de pontos do filtro atual;
//   - ranking de ATs em uma linha por AT (cabem ~2x mais ATs) e com a mesma
//     escala das outras barras (traco = meta de 100%);
//   - alerta < 80% sem fundo rosa nem luz piscando: cor so no selo, na barra
//     e em um filete a esquerda (menos ruido quando ha muitos alertas);
//   - botao "?" (passe o mouse ou clique) explica a regra de pontos e a escala;
//   - textos pequenos com contraste maior; barras animam ao trocar o filtro.
//
// COMPATIBILIDADE COM O HTML CONTENT (o visual nao executa JavaScript)
//   - so HTML + CSS: grid/flex, variaveis CSS, position:sticky, :hover,
//     :focus-within, @keyframes, @media e SVG embutido;
//   - nao usa <input>/<label>/<button> nem fontes, imagens ou CSS externos
//     (a edicao certificada do visual remove esses itens);
//   - atributo style so carrega variaveis (--w, --p) e fica em um <div class='v'>
//     neutro; cores e fontes ficam sempre no bloco <style>;
//   - 100vh de proposito: o visual envolve o conteudo em um wrapper sem altura;
//   - nao usar aspas duplas dentro do CSS/HTML (quebram a string DAX).
// ============================================================================

    )
VAR _pontosIFI7Prata1Atual =
    CALCULATE(
        [BSC Pontos Reais],
        REMOVEFILTERS( 'Seletor Cor' ), REMOVEFILTERS( 'Seletor Indicador BSC' ),
        REMOVEFILTERS( dAT[AREA ] ), REMOVEFILTERS( dAT[COR_AREA ] ),
        TREATAS( { "PRATA 1" }, dAT[COR_AREA ] ),

VAR _resultadoIFI7Prata2 =
    CALCULATE(
        [BSC Resultado Selecionado],

                TREATAS( { _ind }, 'Seletor Indicador BSC'[Indicador] )
            )
    )
        + _ajusteIFI7Ouro

    SUMX(
        _indicadores,
        VAR _ind = [Indicador]

                REMOVEFILTERS( 'Seletor Cor' ),
                REMOVEFILTERS( 'Seletor Indicador BSC' ),
                REMOVEFILTERS( dAT[AREA ] ),
                REMOVEFILTERS( dAT[COR_AREA ] ),
                TREATAS( { "BRONZE" }, dAT[COR_AREA ] ),
                TREATAS( { _ind }, 'Seletor Indicador BSC'[Indicador] )

    )

VAR _pontosPrata1 =
    SUMX(
        _indicadores,
        VAR _ind = [Indicador]
        RETURN
            CALCULATE(
                [BSC Pontos Reais],
                REMOVEFILTERS( 'Seletor Cor' ),
                REMOVEFILTERS( 'Seletor Indicador BSC' ),

                REMOVEFILTERS( dAT[AREA ] ),
                REMOVEFILTERS( dAT[COR_AREA ] ),
                TREATAS( { "PRATA 2" }, dAT[COR_AREA ] ),
                TREATAS( { _ind }, 'Seletor Indicador BSC'[Indicador] )
            )

        + _ajusteIFI7Prata2
VAR _atingGeral = DIVIDE( _pontosGeral, _totalPontosDisponiveis )

// ============================================================================
// RANKING DINÂMICO DOS QUATRO CARDS DE FAIXA
// ============================================================================
VAR _rankingBase =
    UNION(
        ROW("Cor", "OURO", "Pontos", COALESCE(_pontosOuro, 0)),
        ROW("Cor", "BRONZE", "Pontos", COALESCE(_pontosBronze, 0)),
        ROW("Cor", "PRATA 1", "Pontos", COALESCE(_pontosPrata1, 0)),
        ROW("Cor", "PRATA 2", "Pontos", COALESCE(_pontosPrata2, 0))
    )
VAR _rankingCards =
    ADDCOLUMNS(
        _rankingBase,
        "Posicao",
            VAR _corAtual = [Cor]
            VAR _pontosAtual = [Pontos]
            RETURN
                1 + COUNTROWS(
                    FILTER(
                        _rankingBase,
                        [Pontos] > _pontosAtual
                            || ([Pontos] = _pontosAtual && [Cor] < _corAtual)
                    )
                )
    )

// ============================================================================
// APRESENTACAO - CARDS DE FAIXA
// Escala de cor do atingimento (a mesma da tabela e do ranking):
//   green >= 100% | yellow >= 90% | orange >= 80% | red < 80% | neutral = sem dado
// ============================================================================
VAR _faixaGeral =
    SWITCH(
        TRUE(),
        ISBLANK( _atingGeral ), "neutral",
        _atingGeral >= 1, "green",
        _atingGeral >= 0.90, "yellow",
        _atingGeral >= 0.80, "orange",
        "red"
    )
VAR _faltaGeral = _totalPontosDisponiveis - COALESCE( _pontosGeral, 0 )
VAR _rodapeGeral =
    SWITCH(
        TRUE(),
        ISBLANK( _pontosGeral ), "sem pontuação neste filtro",
        _faltaGeral > 0, "<b>" & FORMAT( _faltaGeral, "0.00", _loc ) & " pts</b> abaixo da meta",
        "<b>" & FORMAT( - _faltaGeral, "0.00", _loc ) & " pts</b> acima da meta"
    )
VAR _cardGeral =
    "<div class='v' style='--p:" & FORMAT( MIN( COALESCE( _atingGeral, 0 ), 1 ) * 100, "0.0", "en-US" ) & "'>" &
    "<div class='card hero c-geral" & IF( _corSelecionada = "GERAL", " active", "" ) & "'" &
        " title='GERAL: " & FORMAT( _pontosGeral, "0.00", _loc ) & " pts de " & FORMAT( _totalPontosDisponiveis, "0", _loc ) &
        " disponíveis (" & FORMAT( _atingGeral, "0.0%", _loc ) & "). Anel fechado = meta de 100%.'>" &
        "<div class='hero-txt'>" &
            "<div class='card-top'><i class='sw'></i><span class='card-name'>GERAL</span>" &
                "<span class='card-sub'>" & FORMAT( _qtdATGeral, "0", _loc ) & IF( _qtdATGeral = 1, " AT", " ATs" ) & "</span></div>" &
            "<div class='card-main'><span class='score'>" & IF( ISBLANK( _pontosGeral ), "—", FORMAT( _pontosGeral, "0.00", _loc ) ) &
                "<small>de " & FORMAT( _totalPontosDisponiveis, "0", _loc ) & " pts</small></span></div>" &
            "<div class='card-foot'><span>" & _rodapeGeral & "</span></div>" &
        "</div>" &
        "<div class='gauge " & _faixaGeral & "'>" &
            "<svg class='ring' viewBox='0 0 36 36' aria-hidden='true'>" &
                "<circle class='ring-t' cx='18' cy='18' r='15.9155'></circle>" &
                "<circle class='ring-f' cx='18' cy='18' r='15.9155'></circle></svg>" &
            "<span class='gauge-v'>" & IF( ISBLANK( _atingGeral ), "—", FORMAT( _atingGeral, "0%", _loc ) ) & "</span>" &
        "</div>" &
    "</div></div>"

// Um unico molde para os 4 cards de faixa, na ordem do ranking dinamico.
VAR _cardsCoresOrdenados =
    CONCATENATEX(
        _rankingCards,
        VAR _cor = [Cor]
        VAR _pts = [Pontos]
        VAR _pos = [Posicao]
        VAR _classeCor =
            SWITCH( _cor, "OURO", "c-ouro", "BRONZE", "c-bronze", "PRATA 1", "c-prata1", "PRATA 2", "c-prata2", "c-geral" )
        VAR _qtd =
            SWITCH( _cor, "OURO", _qtdATOuro, "BRONZE", _qtdATBronze, "PRATA 1", _qtdATPrata1, "PRATA 2", _qtdATPrata2 )
        VAR _ating = DIVIDE( _pts, _totalPontosDisponiveis )
        // o ranking das faixas troca "sem pontuacao" por 0: aqui volta a ser "sem dados"
        VAR _semPontos = ( _pts = 0 )
        VAR _faixa =
            SWITCH(
                TRUE(),
                _semPontos || ISBLANK( _ating ), "neutral",
                _ating >= 1, "green",
                _ating >= 0.90, "yellow",
                _ating >= 0.80, "orange",
                "red"
            )
        VAR _delta = _pts - COALESCE( _pontosGeral, 0 )
        RETURN
            "<div class='v' style='--w:" & FORMAT( MIN( COALESCE( _ating, 0 ), _tetoBarra ) / _tetoBarra * 100, "0.0", "en-US" ) & "%'>" &
            "<div class='card " & _classeCor & IF( _corSelecionada = _cor, " active", "" ) & "'" &
                " title='" & _cor & ": " & FORMAT( _pos, "0", _loc ) & "º lugar entre as faixas · " & FORMAT( _pts, "0.00", _loc ) &
                " pts (" & FORMAT( _ating, "0.0%", _loc ) & ") · " & FORMAT( _delta, "+0.00;-0.00;0.00", _loc ) &
                " pts em relação ao Geral. O traço na barra marca a meta de 100%.'>" &
                "<div class='card-top'><i class='sw'></i><span class='card-name'>" & _cor & "</span>" &
                    "<span class='card-sub'>" & FORMAT( _qtd, "0", _loc ) & IF( _qtd = 1, " AT", " ATs" ) & "</span>" &
                    "<b class='rank'>" & FORMAT( _pos, "0", _loc ) & "º</b></div>" &
                "<div class='card-main'><span class='score'>" & IF( _semPontos, "—", FORMAT( _pts, "0.00", _loc ) ) & "<small>pts</small></span></div>" &
                "<div class='track'><u></u></div>" &
                "<div class='card-foot'>" &
                    IF(
                        _semPontos,
                        "<span class='pill neutral'>sem dados</span>",
                        "<span class='delta " & IF( _delta >= 0, "up", "down" ) & "'>" &
                            FORMAT( _delta, "+0.00;-0.00;0.00", _loc ) & "</span><span class='vs'>vs geral</span>" &
                            "<span class='pill " & _faixa & "'>" & IF( _ating >= 1, "▲ ", "▼ " ) & FORMAT( _ating, "0.0%", _loc ) & "</span>"
                    ) &
                "</div>" &
            "</div></div>",
        "",
        [Posicao], ASC
    )

// ============================================================================
// CALCULO (inalterado) - TABELA DOS 12 INDICADORES
// Monta primeiro uma tabela com os valores de cada indicador; subtotais,
// contadores da legenda e linhas exibidas saem todos da mesma fonte.
// ============================================================================
VAR _tabIndBase =
    ADDCOLUMNS(
        _indicadores,
        "__ResultadoBruto",
            VAR _ind = [Indicador]
            RETURN
                IF(
                    [AnaliticoAT] = 1,
                    CALCULATE(
                        [BSC Resultado Selecionado],
                        REMOVEFILTERS( 'Seletor Cor' ),
                        REMOVEFILTERS( 'Seletor Indicador BSC' ),
                        FILTER(
                            ALL( dAT[COR_AREA ] ),
                            _corSelecionada = "GERAL"
                                || dAT[COR_AREA ] = _corSelecionada
                        ),
                        TREATAS( { _ind }, 'Seletor Indicador BSC'[Indicador] )
                    ),
                    CALCULATE(
                        [BSC Resultado Selecionado],
                        REMOVEFILTERS( 'Seletor Cor' ),
                        REMOVEFILTERS( 'Seletor Indicador BSC' ),
