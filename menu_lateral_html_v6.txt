// ============================================================================
// MENU LATERAL  (v6 - refeito do zero)   |   visual 2 de 3
// Fica "por baixo" dos botoes nativos transparentes do Power BI, que continuam
// fazendo a navegacao (o HTML Content nao navega entre paginas).
//
// EM CADA PAGINA altere somente _paginaAtual para um destes valores:
// "BSC", "IFI", "IRR", "TDNA Instalação", "TDNA Reparos" ou "Abertura Hierarquia".
// Para incluir, renomear ou reordenar paginas, edite a tabela _paginas.
//
// PALETA: troque so a palavra em _tema ->  "OCEANO" | "AURORA" | "GRAFITE"
// (use o mesmo tema do corpo e da atualizacao das bases).
//
// >>> BOTOES NATIVOS CONTINUAM ALINHADOS <<<
// O 1o item comeca a 109px do topo do visual, cada item tem 36px de altura,
// 7px de intervalo e 14px de margem lateral: exatamente como no menu atual.
// Agora essas medidas sao fixas no CSS (antes dependiam da fonte).
//
// O QUE TEM DE NOVO
//   - visual escuro novo, com a mesma paleta da faixa de destaque do corpo;
//   - marca com monograma; icone por pagina; pagina atual com barra luminosa,
//     icone aceso e um feixe de luz passando (a unica animacao continua);
//   - bloco CORES: as 4 cores com a quantidade de ATs de cada uma; a cor
//     filtrada fica acesa e as outras apagam;
//   - bloco FILTRO ATUAL: cluster + etiquetas da cor e da AT selecionadas;
//   - aviso sobre filtros em um cartao discreto, com icone de atencao.
//
// DICA: o botao nativo fica por cima, entao o HTML nao "sente" o mouse. Para
// dar retorno ao passar o mouse, no botao nativo use Estilo > Estado
// "Ao passar o mouse" > Preenchimento branco com ~90% de transparencia.
// Nao usar aspas duplas dentro do CSS/HTML (quebram a string DAX).
// ============================================================================

VAR _paginaAtual =
    "BSC"

// ------------------------------------------------------------------- PALETA
VAR _tema = "OCEANO"

VAR _tokens =
    SWITCH(
        _tema,
        "AURORA",
            "--d0:#140A2E;--d1:#22114F;--d2:#43208A;--a1:#F472B6;--a2:#A78BFA;--g1:rgba(244,114,182,.24);--g2:rgba(167,139,250,.30)",
        "GRAFITE",
            "--d0:#0C1016;--d1:#161C26;--d2:#2B3442;--a1:#7DD3FC;--a2:#CBD5E1;--g1:rgba(125,211,252,.14);--g2:rgba(148,163,184,.20)",
        "--d0:#050E26;--d1:#0A1A42;--d2:#123A7C;--a1:#22D3EE;--a2:#60A5FA;--g1:rgba(34,211,238,.22);--g2:rgba(59,130,246,.30)"
    )

// ------------------------------------------------------------- TEXTOS FIXOS
VAR _marca = "Indicadores"
VAR _sigla = "ARF"
VAR _subtitulo = "BSC OPERACIONAL"
VAR _cluster = "PERUS / BUTANTÃ"
VAR _avisoTitulo = "Aviso sobre filtros"
VAR _avisoTexto =
    "Os indicadores IRT, Indisponibilidade Móvel e Indisponibilidade FTTH não respeitarão os filtros por enquanto, pois não dispõem de analítico disponível no ELOS."

// --------------------------------------------------------- CONTEXTO (FILTROS)
VAR _corSelecionada =
    SELECTEDVALUE(
        'Seletor Cor'[COR_AREA],
        "GERAL"
    )

VAR _areaSelecionada =
    SELECTEDVALUE(
        dAT[AREA ]
    )

VAR _temAreaSelecionada =
    HASONEVALUE(
        dAT[AREA ]
    )

// Quantidade de ATs por cor: mesmo calculo usado no corpo do BSC
VAR _qtdATGeral =
    CALCULATE(
        DISTINCTCOUNT( dAT[AREA ] ),
        REMOVEFILTERS( 'Seletor Cor' ),
        REMOVEFILTERS( dAT[AREA ] ),
        REMOVEFILTERS( dAT[COR_AREA ] )
    )
VAR _qtdATOuro =
    CALCULATE(
        DISTINCTCOUNT( dAT[AREA ] ),
        REMOVEFILTERS( 'Seletor Cor' ),
        REMOVEFILTERS( dAT[AREA ] ),
        REMOVEFILTERS( dAT[COR_AREA ] ),
        TREATAS( { "OURO" }, dAT[COR_AREA ] )
    )
VAR _qtdATBronze =
    CALCULATE(
        DISTINCTCOUNT( dAT[AREA ] ),
        REMOVEFILTERS( 'Seletor Cor' ),
        REMOVEFILTERS( dAT[AREA ] ),
        REMOVEFILTERS( dAT[COR_AREA ] ),
        TREATAS( { "BRONZE" }, dAT[COR_AREA ] )
    )
VAR _qtdATPrata1 =
    CALCULATE(
        DISTINCTCOUNT( dAT[AREA ] ),
        REMOVEFILTERS( 'Seletor Cor' ),
        REMOVEFILTERS( dAT[AREA ] ),
        REMOVEFILTERS( dAT[COR_AREA ] ),
        TREATAS( { "PRATA 1" }, dAT[COR_AREA ] )
    )
VAR _qtdATPrata2 =
    CALCULATE(
        DISTINCTCOUNT( dAT[AREA ] ),
        REMOVEFILTERS( 'Seletor Cor' ),
        REMOVEFILTERS( dAT[AREA ] ),
        REMOVEFILTERS( dAT[COR_AREA ] ),
        TREATAS( { "PRATA 2" }, dAT[COR_AREA ] )
    )
VAR _cores =
    UNION(
        ROW( "Cor", "OURO", "Ordem", 1, "Classe", "c-ouro", "Qtd", _qtdATOuro ),
        ROW( "Cor", "PRATA 1", "Ordem", 2, "Classe", "c-prata1", "Qtd", _qtdATPrata1 ),
        ROW( "Cor", "PRATA 2", "Ordem", 3, "Classe", "c-prata2", "Qtd", _qtdATPrata2 ),
        ROW( "Cor", "BRONZE", "Ordem", 4, "Classe", "c-bronze", "Qtd", _qtdATBronze )
    )
VAR _corEhFaixa =
    _corSelecionada = "OURO" || _corSelecionada = "BRONZE"
        || _corSelecionada = "PRATA 1" || _corSelecionada = "PRATA 2"
VAR _classeCorSel =
    SWITCH( _corSelecionada, "OURO", "c-ouro", "BRONZE", "c-bronze", "PRATA 1", "c-prata1", "PRATA 2", "c-prata2", "c-geral" )
VAR _chipsCores =
    CONCATENATEX(
        _cores,
        "<div class='fx " & [Classe] & IF( [Cor] = _corSelecionada, " on", "" ) & "'>" &
            "<i class='sw'></i><b>" & [Cor] & "</b><small>" & FORMAT( [Qtd], "0" ) & "</small></div>",
        "",
        [Ordem], ASC
    )

// --------------------------------------------------------------------- PAGINAS
// Pagina = texto exibido (e o valor comparado com _paginaAtual).
// Icone  = miolo de um SVG 24x24 desenhado so com traco.
VAR _paginas =
    DATATABLE(
        "Pagina", STRING,
        "Ordem", INTEGER,
        "Icone", STRING,
        {
            { "BSC", 1, "<rect x='3' y='3' width='7' height='9' rx='1.5'></rect><rect x='14' y='3' width='7' height='5' rx='1.5'></rect><rect x='14' y='12' width='7' height='9' rx='1.5'></rect><rect x='3' y='16' width='7' height='5' rx='1.5'></rect>" },
            { "IFI", 2, "<path d='M3 12h4l3-8 4 16 3-8h4'></path>" },
            { "IRR", 3, "<path d='M17 2l4 4-4 4'></path><path d='M3 11V9a3 3 0 0 1 3-3h15'></path><path d='M7 22l-4-4 4-4'></path><path d='M21 13v2a3 3 0 0 1-3 3H3'></path>" },
            { "TDNA Instalação", 4, "<path d='M3 10.5 12 3l9 7.5'></path><path d='M5 9.5V20h14V9.5'></path><path d='M10 20v-6h4v6'></path>" },
            { "TDNA Reparos", 5, "<path d='M14.7 6.3a1 1 0 0 0 0 1.4l1.6 1.6a1 1 0 0 0 1.4 0l3.77-3.77a6 6 0 0 1-7.94 7.94l-6.91 6.91a2.12 2.12 0 0 1-3-3l6.91-6.91a6 6 0 0 1 7.94-7.94l-3.76 3.76z'></path>" },
            { "Abertura Hierarquia", 6, "<rect x='9' y='2' width='6' height='5' rx='1'></rect><rect x='2' y='17' width='6' height='5' rx='1'></rect><rect x='16' y='17' width='6' height='5' rx='1'></rect><path d='M12 7v5M5 17v-3a2 2 0 0 1 2-2h10a2 2 0 0 1 2 2v3'></path>" }
        }
    )

VAR _itens =
    CONCATENATEX(
        _paginas,
        VAR _ativa = ( [Pagina] = _paginaAtual )
        RETURN
            "<div class='it" & IF( _ativa, " on", "" ) & "'>" &
                IF( _ativa, "<span class='scan'></span>", "" ) &
                "<span class='ic-w'><svg class='ic' viewBox='0 0 24 24' aria-hidden='true'>" & [Icone] & "</svg></span>" &
                "<span class='lb'>" & [Pagina] & "</span>" &
                IF( _ativa, "<i class='live'></i>", "" ) &
            "</div>",
        "",
        [Ordem], ASC
    )
