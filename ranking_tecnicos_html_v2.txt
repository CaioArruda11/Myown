Ranking Tecnicos HTML = 

// ============================================================================
// RANKING DE TECNICOS / SUPERVISORES  (v2 - refeito do zero)
// Pagina "Abertura Hierarquia". HTML Content ocupando o corpo da pagina.
// Fundo TRANSPARENTE. Mesma linguagem visual do BSC v5.
//
// PALETA: troque so a palavra em _tema ->  "GRAFITE" | "OCEANO" | "AURORA"
// (use o mesmo tema do menu lateral desta pagina).
//
// REGRAS DE CALCULO: as mesmas da versao anterior. Os blocos "CALCULO
// (inalterado)" foram copiados sem nenhuma alteracao: configuracao de pontos,
// metas e volume minimo, bases, grade, ranking e indicadores do topo.
// Para ajustar pontos / metas / volume minimo, edite a tabela _cfg, como antes.
//
// O QUE TEM DE NOVO
//   1. FAIXA DE DESTAQUE escura no topo, que acompanha o filtro:
//        - velocimetro do atingimento (visao geral, tecnico ou supervisor
//          selecionado), com zonas, traco da meta e a media como referencia;
//        - os 5 indicadores da selecao em grafico de bala: real, meta,
//          atingimento, pontos e a variacao contra o mes anterior. Isso junta
//          em um bloco so os 3 cards do topo e o painel "Detalhe" antigos;
//        - resumo do grupo + HISTOGRAMA do atingimento (quantos em cada faixa).
//   2. RANKING na largura toda (o painel lateral saiu): nome completo,
//      supervisor, celulas coloridas por indicador, COBERTURA (quantos dos 5
//      indicadores tem dado), pontos em barra com traco da meta e atingimento.
//   3. BOTOES clicaveis: "Melhores | Piores" inverte a ordem da lista;
//      "Todos | Criticos" mostra so quem esta abaixo de 80%.
//   4. A linha selecionada no filtro fica presa no topo/rodape ao rolar.
//   5. Sem "-nan(ind)": divisao por zero vira "sem dados". Sem pisca-pisca.
//   6. DIAGNOSTICO DE "SEM DADOS" (v2.1): quando um indicador nao tem resultado,
//      a tela diz o motivo ("sem altas", "sem reparos", "volume baixo") e avisa
//      quando NINGUEM do filtro tem dado daquele indicador - sinal de que a
//      base do indicador nao tem linhas para o mes/filtro escolhido.
//
// COMPATIBILIDADE COM O HTML CONTENT (nao executa JavaScript)
//   - so HTML + CSS + SVG embutido; botoes com <details> + :has(), como nas
//     paginas IFI/IRR e no BSC v5;
//   - atributo style so carrega variaveis CSS e fica em um <div class='v'>
//     neutro; nomes de pessoas nunca entram em atributos (title), so em texto;
//   - 100vh de proposito: o visual envolve o conteudo em um wrapper sem altura;
//   - nao usar aspas duplas dentro do CSS/HTML (quebram a string DAX).
// ============================================================================

// ------------------------------------------------------------------- PALETA
VAR _tema = "GRAFITE"

// d0/d1/d2 = fundos escuros | a1/a2 = cores de destaque | g1/g2 = brilhos
// pri/pri-b = destaque sobre fundo claro
VAR _tokens =
    SWITCH(
        _tema,
        "AURORA",
            "--d0:#140A2E;--d1:#22114F;--d2:#43208A;--a1:#F472B6;--a2:#A78BFA;--g1:rgba(244,114,182,.24);--g2:rgba(167,139,250,.30);--pri:#7C3AED;--pri-b:#F3EDFF",
        "OCEANO",
            "--d0:#050E26;--d1:#0A1A42;--d2:#123A7C;--a1:#22D3EE;--a2:#60A5FA;--g1:rgba(34,211,238,.22);--g2:rgba(59,130,246,.30);--pri:#2563EB;--pri-b:#E9F0FF",
        "--d0:#0C1016;--d1:#161C26;--d2:#2B3442;--a1:#7DD3FC;--a2:#CBD5E1;--g1:rgba(125,211,252,.14);--g2:rgba(148,163,184,.20);--pri:#334155;--pri-b:#EEF2F6"
    )

// ============================================================================
// CALCULO (inalterado) - CONFIGURACAO, SELECOES, BASES, GRADE, RANKING E
// INDICADORES DO TOPO
// ============================================================================
VAR _loc = "pt-BR"
VAR _visao = SELECTEDVALUE( 'Seletor Visão'[Visao], "TÉCNICO" )
VAR _ehSupervisor = _visao = "SUPERVISOR"
VAR _rotuloEnt = IF( _ehSupervisor, "Supervisores", "Técnicos" )

// ----------------------------------------------------------------------------
// CONFIGURAÇÃO  (AJUSTE PONTOS, METAS E VOLUME MÍNIMO AQUI)
// MenorMelhor = 1 -> Meta / Real   |   MenorMelhor = 0 -> Real / Meta
// VolMin = volume mínimo (altas/reparos) para o indicador contar.
//          0 = não checa volume. Quando as medidas de volume estiverem
//          certas, coloque 3 (ou o mínimo que fizer sentido) em IFI/IRR.
// ----------------------------------------------------------------------------
VAR _cfg =
    DATATABLE(
        "Indicador", STRING,
        "NomeTela", STRING,
        "Sigla", STRING,
        "Ordem", INTEGER,
        "Pontos", DOUBLE,
        "Meta", DOUBLE,
        "MenorMelhor", INTEGER,
        "Tipo", STRING,
        "VolMin", INTEGER,
        "RotDen", STRING,
        "RotNum", STRING,
        {
            { "IFI",              "IFI FTTH",          "IFI",    1, 20, 0.040, 1, "%",   0, "Altas",        "Reparos" },
            { "IFI 7 DIAS",       "IFI FTTH 7 Dias",   "IFI 7D", 2, 20, 0.020, 1, "%",   0, "Altas",        "Reparos" },
            { "IRR",              "Repetidas FTTH",    "IRR",    3, 20, 0.100, 1, "%",   0, "Reparos",      "Repetidas" },
            { "IRR 7 DIAS",       "Repetidas 7 Dias",  "IRR 7D", 4, 20, 0.050, 1, "%",   0, "Reparos",      "Repetidas" },
            { "PRODUTIVIDADE PU", "Produtividade PU",  "PU",     5, 20, 3.500, 0, "Qtd", 0, "Reparos",      "Instalações" }
        }
    )
// RotDen / RotNum = rótulos do tooltip (denominador / numerador).
// O volume mínimo (VolMin) agora é checado contra o DENOMINADOR (altas p/ IFI, reparos p/ IRR).
VAR _qtdIndicadores = COUNTROWS( _cfg )
VAR _ptsDisp = SUMX( _cfg, [Pontos] )

// ----------------------------------------------------------------------------
// SELEÇÕES
// ----------------------------------------------------------------------------
VAR _tecSel =
    IF(
        ISFILTERED( 'dMO X CONTRATO'[NOME] ) && HASONEVALUE( 'dMO X CONTRATO'[NOME] ),
        VALUES( 'dMO X CONTRATO'[NOME] )
    )
VAR _supSel =
    IF(
        ISFILTERED( 'dMO X CONTRATO'[SUPERVISOR] ) && HASONEVALUE( 'dMO X CONTRATO'[SUPERVISOR] ),
        VALUES( 'dMO X CONTRATO'[SUPERVISOR] )
    )
VAR _entSel = IF( _ehSupervisor, _supSel, _tecSel )
VAR _entDetalhe = IF( ISBLANK( _entSel ), "__GERAL__", _entSel )

VAR _supsContexto =
    CALCULATETABLE(
        VALUES( 'dMO X CONTRATO'[SUPERVISOR] ),
        REMOVEFILTERS( 'dMO X CONTRATO'[NOME] )
    )

// ----------------------------------------------------------------------------
// BASE: mesmo mecanismo de uma tabela nativa (linha a linha por NOME/SUPERVISOR)
// As medidas do usuário entram aqui (ver lista no topo do arquivo)
// ----------------------------------------------------------------------------
VAR _baseTec =
    CALCULATETABLE(
        SELECTCOLUMNS(
            ADDCOLUMNS(
                VALUES( 'dMO X CONTRATO'[NOME] ),
                "@IFI", [% IFI], "@IFI7", [% IFI 7D], "@IRR", [IRR DAT_FECHAMENTO], "@IRR7", [% IRR CAMPO 7D], "@PU", [PU],
                "@nIFI",  [Contagem Reparos E],      "@dIFI",  [Contagem Alta Enc.],
                "@nIFI7", [Contagem Reparos E 7D],   "@dIFI7", [Contagem Alta Enc. 7D],
                "@nIRR",  [Qtd Repetidas IRR],       "@dIRR",  [Qtd Reparos IRR],
                "@nIRR7", [Qtd Repetidas 7D Campo],  "@dIRR7", [Qtd Reparos 7D Campo],
                "@nPU",   [PU Qtd Instalacoes],      "@dPU",   [PU Qtd Reparos]
            ),
            "Entidade", 'dMO X CONTRATO'[NOME],
            "@IFI", [@IFI], "@IFI7", [@IFI7], "@IRR", [@IRR], "@IRR7", [@IRR7], "@PU", [@PU],
            "@nIFI", [@nIFI], "@dIFI", [@dIFI], "@nIFI7", [@nIFI7], "@dIFI7", [@dIFI7],
            "@nIRR", [@nIRR], "@dIRR", [@dIRR], "@nIRR7", [@nIRR7], "@dIRR7", [@dIRR7],
            "@nPU", [@nPU], "@dPU", [@dPU]
        ),
        REMOVEFILTERS( 'Seletor Visão' ),
        FILTER( ALL( 'dMO X CONTRATO'[NOME] ), NOT _ehSupervisor )
    )

VAR _baseSup =
    CALCULATETABLE(
        SELECTCOLUMNS(
            ADDCOLUMNS(
                VALUES( 'dMO X CONTRATO'[SUPERVISOR] ),
                "@IFI", [% IFI], "@IFI7", [% IFI 7D], "@IRR", [IRR DAT_FECHAMENTO], "@IRR7", [% IRR TOTAL 7D], "@PU", [PU],
                "@nIFI",  [Contagem Reparos E],      "@dIFI",  [Contagem Alta Enc.],
                "@nIFI7", [Contagem Reparos E 7D],   "@dIFI7", [Contagem Alta Enc. 7D],
                "@nIRR",  [Qtd Repetidas IRR],       "@dIRR",  [Qtd Reparos IRR],
                "@nIRR7", [Qtd Repetidas 7D Total],  "@dIRR7", [Qtd Reparos 7D Total],
                "@nPU",   [PU Qtd Instalacoes],      "@dPU",   [PU Qtd Reparos]
            ),
            "Entidade", 'dMO X CONTRATO'[SUPERVISOR],
            "@IFI", [@IFI], "@IFI7", [@IFI7], "@IRR", [@IRR], "@IRR7", [@IRR7], "@PU", [@PU],
            "@nIFI", [@nIFI], "@dIFI", [@dIFI], "@nIFI7", [@nIFI7], "@dIFI7", [@dIFI7],
            "@nIRR", [@nIRR], "@dIRR", [@dIRR], "@nIRR7", [@nIRR7], "@dIRR7", [@dIRR7],
            "@nPU", [@nPU], "@dPU", [@dPU]
        ),
        REMOVEFILTERS( 'Seletor Visão' ),
        REMOVEFILTERS( 'dMO X CONTRATO'[NOME] ),
        FILTER( ALL( 'dMO X CONTRATO'[SUPERVISOR] ), _ehSupervisor )
    )

VAR _baseGeral =
    CALCULATETABLE(
        ROW(
            "Entidade", "__GERAL__",
            "@IFI", [% IFI], "@IFI7", [% IFI 7D], "@IRR", [IRR DAT_FECHAMENTO], "@IRR7", [% IRR TOTAL 7D], "@PU", [PU],
            "@nIFI",  [Contagem Reparos E],      "@dIFI",  [Contagem Alta Enc.],
            "@nIFI7", [Contagem Reparos E 7D],   "@dIFI7", [Contagem Alta Enc. 7D],
            "@nIRR",  [Qtd Repetidas IRR],       "@dIRR",  [Qtd Reparos IRR],
            "@nIRR7", [Qtd Repetidas 7D Total],  "@dIRR7", [Qtd Reparos 7D Total],
            "@nPU",   [PU Qtd Instalacoes],      "@dPU",   [PU Qtd Reparos]
        ),
        REMOVEFILTERS( 'Seletor Visão' ),
        REMOVEFILTERS( 'dMO X CONTRATO'[NOME] ),
        FILTER(
            ALL( 'dMO X CONTRATO'[SUPERVISOR] ),
            _ehSupervisor || CONTAINSROW( _supsContexto, 'dMO X CONTRATO'[SUPERVISOR] )
        )
    )

VAR _base = UNION( _baseTec, _baseSup, _baseGeral )

VAR _entidades =
    FILTER(
        SELECTCOLUMNS( _base, "Entidade", [Entidade] ),
        [Entidade] <> "__GERAL__" && [Entidade] <> ""
    )

// ----------------------------------------------------------------------------
// GRADE ENTIDADE x INDICADOR (sem recalcular medidas)
// ----------------------------------------------------------------------------
VAR _grade1 =
    ADDCOLUMNS(
        CROSSJOIN( _base, _cfg ),
        "__ResBruto",
            SWITCH(
                [Indicador],
                "IFI", [@IFI],
                "IFI 7 DIAS", [@IFI7],
                "IRR", [@IRR],
                "IRR 7 DIAS", [@IRR7],
                "PRODUTIVIDADE PU", [@PU]
            ),
        "__Num",
            SWITCH(
                [Indicador],
                "IFI", [@nIFI],
                "IFI 7 DIAS", [@nIFI7],
                "IRR", [@nIRR],
                "IRR 7 DIAS", [@nIRR7],
                "PRODUTIVIDADE PU", [@nPU]
            ),
        "__Den",
            SWITCH(
                [Indicador],
                "IFI", [@dIFI],
                "IFI 7 DIAS", [@dIFI7],
                "IRR", [@dIRR],
                "IRR 7 DIAS", [@dIRR7],
                "PRODUTIVIDADE PU", [@dPU]
            )
    )
VAR _grade2 =
    ADDCOLUMNS(
        _grade1,
        "__Res",
            VAR _r = [__ResBruto]
            VAR _txt = FORMAT( _r, "0.000000", "en-US" )
            VAR _nan = SEARCH( "n", _txt, 1, 0 ) > 0
            VAR _semVolume = [VolMin] > 0 && COALESCE( [__Den], 0 ) < [VolMin]
            RETURN IF( _nan || _semVolume, BLANK(), _r )
    )
VAR _grade3 =
    ADDCOLUMNS(
        _grade2,
        "__Ating",
            VAR _r = [__Res]
            VAR _m = [Meta]
            RETURN
                SWITCH(
                    TRUE(),
                    ISBLANK( _r ) || ISBLANK( _m ), BLANK(),
                    [MenorMelhor] = 1 && _r = 0, 1.30,
                    [MenorMelhor] = 1, DIVIDE( _m, _r ),
                    DIVIDE( _r, _m )
                )
    )
VAR _grade =
    ADDCOLUMNS(
        _grade3,
        "__Pts", IF( ISBLANK( [__Ating] ), BLANK(), MIN( [Pontos] * 1.30, [Pontos] * [__Ating] ) ),
        "__Tier",
            VAR _a = [__Ating]
            RETURN SWITCH( TRUE(), ISBLANK( _a ), "tN", _a >= 1, "tG", _a >= 0.90, "tY", _a >= 0.80, "tO", "tR" ),
        "__Txt",
            IF( ISBLANK( [__Res] ), "—", IF( [Tipo] = "%", FORMAT( [__Res], "0.0%", _loc ), FORMAT( [__Res], "0.0", _loc ) ) ),
        "__MetaTxt",
            IF( [Tipo] = "%", FORMAT( [Meta], "0.0%", _loc ), FORMAT( [Meta], "0.0", _loc ) ),
        "__Larg",
            FORMAT( MIN( COALESCE( [__Ating], 0 ), 1.3 ) / 1.3 * 100, "0.0", "en-US" ),
        "__Tip",
            VAR _d = [__Den]
            VAR _n = [__Num]
            VAR _dTxt = IF( ISBLANK( _d ), "—", FORMAT( _d, "#,0", _loc ) )
            VAR _nTxt = IF( ISBLANK( _n ), "—", FORMAT( _n, "#,0", _loc ) )
            RETURN
                "<span class='tt'>" &
                    "<span><em>" & [RotDen] & "</em><b>" & _dTxt & "</b></span>" &
                    "<span><em>" & [RotNum] & "</em><b>" & _nTxt & "</b></span>" &
                "</span>"
    )

// ----------------------------------------------------------------------------
// RANKING
// ----------------------------------------------------------------------------
VAR _ranking0 =
    ADDCOLUMNS(
        _entidades,
        "__PtsTotal",
            VAR _e = [Entidade]
            RETURN COALESCE( SUMX( FILTER( _grade, [Entidade] = _e ), [__Pts] ), 0 ),
        "__QtdInd",
            VAR _e = [Entidade]
            RETURN COALESCE( COUNTROWS( FILTER( _grade, [Entidade] = _e && NOT ISBLANK( [__Res] ) ) ), 0 ),
        "__PtsDisp",
            VAR _e = [Entidade]
            RETURN
                COALESCE(
                    SUMX( FILTER( _grade, [Entidade] = _e && NOT ISBLANK( [__Res] ) ), [Pontos] ),
                    0
                )
    )
VAR _validos0 = FILTER( _ranking0, [__QtdInd] > 0 )
VAR _validos1 =
    ADDCOLUMNS( _validos0, "__AtingTot", DIVIDE( [__PtsTotal], [__PtsDisp] ) )
VAR _validos =
    ADDCOLUMNS(
        _validos1,
        "__Seq",
            VAR _a = [__AtingTot]
            VAR _e = [Entidade]
            RETURN
                1 + COALESCE(
                    COUNTROWS( FILTER( _validos1, [__AtingTot] > _a || ( [__AtingTot] = _a && [Entidade] < _e ) ) ),
                    0
                ),
        "__Sub",
            VAR _e = [Entidade]
            RETURN
                IF(
                    _ehSupervisor,
                    FORMAT(
                        CALCULATE(
                            DISTINCTCOUNT( 'dMO X CONTRATO'[NOME] ),
                            REMOVEFILTERS( 'dMO X CONTRATO'[NOME] ),
                            REMOVEFILTERS( 'dMO X CONTRATO'[SUPERVISOR] ),
                            TREATAS( { _e }, 'dMO X CONTRATO'[SUPERVISOR] )
                        ),
                        "0"
                    ) & " técnicos",
                    CALCULATE(
                        CONCATENATEX( VALUES( 'dMO X CONTRATO'[SUPERVISOR] ), 'dMO X CONTRATO'[SUPERVISOR], " / " ),
                        REMOVEFILTERS( 'dMO X CONTRATO'[NOME] ),
                        REMOVEFILTERS( 'dMO X CONTRATO'[SUPERVISOR] ),
                        TREATAS( { _e }, 'dMO X CONTRATO'[NOME] )
                    )
                ),
        "__Ini",
            VAR _nm = TRIM( [Entidade] )
            VAR _esp = LEN( _nm ) - LEN( SUBSTITUTE( _nm, " ", "" ) )
            RETURN
                UPPER(
                    LEFT( _nm, 1 )
                        & IF( _esp = 0, "", MID( _nm, SEARCH( "@", SUBSTITUTE( _nm, " ", "@", _esp ) ) + 1, 1 ) )
                ),
        "__Curto",
            VAR _nm = TRIM( [Entidade] )
            VAR _esp = LEN( _nm ) - LEN( SUBSTITUTE( _nm, " ", "" ) )
            RETURN
                IF(
                    _esp = 0,
                    _nm,
                    LEFT( _nm, SEARCH( " ", _nm ) - 1 ) & " "
                        & MID( _nm, SEARCH( "@", SUBSTITUTE( _nm, " ", "@", _esp ) ) + 1, 100 )
                )
    )
VAR _rank =
    ADDCOLUMNS(
        _validos,
        "__TierTot",
            VAR _a = [__AtingTot]
            RETURN SWITCH( TRUE(), _a >= 1, "tG", _a >= 0.90, "tY", _a >= 0.80, "tO", "tR" ),
        "__Cor",
            SWITCH( MOD( [__Seq], 6 ), 0, "#5B6BD6", 1, "#D9678C", 2, "#2BA6A0", 3, "#E0842F", 4, "#7A5CE0", "#3C8DD9" )
    )

VAR _qtd = COALESCE( COUNTROWS( _rank ), 0 )
VAR _mediaAting = AVERAGEX( _rank, [__AtingTot] )
VAR _nG = COALESCE( COUNTROWS( FILTER( _rank, [__TierTot] = "tG" ) ), 0 )
VAR _nY = COALESCE( COUNTROWS( FILTER( _rank, [__TierTot] = "tY" ) ), 0 )
VAR _nO = COALESCE( COUNTROWS( FILTER( _rank, [__TierTot] = "tO" ) ), 0 )
VAR _nR = COALESCE( COUNTROWS( FILTER( _rank, [__TierTot] = "tR" ) ), 0 )

// ----------------------------------------------------------------------------
// LINHA DE CIMA: INDICADORES + RESUMO
// ----------------------------------------------------------------------------
VAR _irr7Atual = IF( _ehSupervisor, [% IRR TOTAL 7D], [% IRR CAMPO 7D] )
VAR _irr7M1 = IF( _ehSupervisor, [% IRR TOTAL 7D M1], [% IRR CAMPO 7D M1] )
VAR _ifi7Atual = [% IFI 7D]
VAR _ifi7M1 = [% IFI 7D M1]
VAR _puAtual = [PU]
VAR _puM1 = [PU M1]

VAR _irr7Status =
    SWITCH( TRUE(), ISBLANK( _irr7Atual ), "neutral", _irr7Atual <= 0.05, "good", _irr7Atual <= DIVIDE( 0.05, 0.90 ), "near", "bad" )
VAR _ifi7Status =
    SWITCH( TRUE(), ISBLANK( _ifi7Atual ), "neutral", _ifi7Atual <= 0.02, "good", _ifi7Atual <= DIVIDE( 0.02, 0.90 ), "near", "bad" )
VAR _puStatus =
    SWITCH( TRUE(), ISBLANK( _puAtual ), "neutral", _puAtual >= 3.5, "good", _puAtual >= 3.5 * 0.90, "near", "bad" )

VAR _irr7Dif = IF( NOT ISBLANK( _irr7Atual ) && NOT ISBLANK( _irr7M1 ), _irr7Atual - _irr7M1 )
VAR _ifi7Dif = IF( NOT ISBLANK( _ifi7Atual ) && NOT ISBLANK( _ifi7M1 ), _ifi7Atual - _ifi7M1 )
VAR _puDif = IF( NOT ISBLANK( _puAtual ) && NOT ISBLANK( _puM1 ), _puAtual - _puM1 )

VAR _irr7Trend = SWITCH( TRUE(), ISBLANK( _irr7Dif ), "neutral", _irr7Dif < 0, "up", _irr7Dif > 0, "down", "flat" )
VAR _ifi7Trend = SWITCH( TRUE(), ISBLANK( _ifi7Dif ), "neutral", _ifi7Dif < 0, "up", _ifi7Dif > 0, "down", "flat" )
VAR _puTrend = SWITCH( TRUE(), ISBLANK( _puDif ), "neutral", _puDif > 0, "up", _puDif < 0, "down", "flat" )

VAR _irr7Seta = SWITCH( _irr7Trend, "up", "▼", "down", "▲", "flat", "●", "—" )
VAR _ifi7Seta = SWITCH( _ifi7Trend, "up", "▼", "down", "▲", "flat", "●", "—" )
VAR _puSeta = SWITCH( _puTrend, "up", "▲", "down", "▼", "flat", "●", "—" )

// ============================================================================
// CALCULO (inalterado) - DETALHE DA SELECAO (visao geral ou entidade filtrada)
// ============================================================================
VAR _ehGeral = _entDetalhe = "__GERAL__"
VAR _detGrade = FILTER( _grade, [Entidade] = _entDetalhe )
VAR _detPts = COALESCE( SUMX( _detGrade, [__Pts] ), 0 )
VAR _detPtsDisp =
    COALESCE( SUMX( FILTER( _detGrade, NOT ISBLANK( [__Res] ) ), [Pontos] ), 0 )
VAR _detAting = DIVIDE( _detPts, _detPtsDisp )
VAR _detQtdInd = COALESCE( COUNTROWS( FILTER( _detGrade, NOT ISBLANK( [__Res] ) ) ), 0 )
VAR _detCor =
    SWITCH( TRUE(), _detAting >= 1, "#10AD68", _detAting >= 0.90, "#D09A12", _detAting >= 0.80, "#E0842F", "#DB3F48" )
VAR _detLinhaRank = FILTER( _rank, [Entidade] = _entDetalhe )
VAR _detTitulo = IF( _ehGeral, "Visão geral", _entDetalhe )
VAR _detSub =
    IF(
        _ehGeral,
        "Todos os " & LOWER( _rotuloEnt ) & " do filtro",
        IF(
            ISEMPTY( _detLinhaRank ),
            "Sem dados no filtro",
            MAXX( _detLinhaRank, [__Seq] ) & "º de " & _qtd & " · " & CONCATENATEX( _detLinhaRank, [__Sub] )
        )
    )
VAR _detAvatar = IF( _ehGeral, "Σ", IF( ISEMPTY( _detLinhaRank ), LEFT( _entDetalhe, 1 ), CONCATENATEX( _detLinhaRank, [__Ini] ) ) )
VAR _detAvCor = IF( _ehGeral, "#7A5CE0", IF( ISEMPTY( _detLinhaRank ), "#5B6BD6", CONCATENATEX( _detLinhaRank, [__Cor] ) ) )

// ============================================================================
// APRESENTACAO - FAIXA DE DESTAQUE
// ============================================================================
VAR _teto = 1.30
VAR _posMeta = FORMAT( 1 / _teto * 100, "0.0", "en-US" )
VAR _pos80 = FORMAT( 0.8 / _teto * 100, "0.0", "en-US" )
VAR _rotuloUm = IF( _ehSupervisor, "Supervisor", "Técnico" )

// As medidas do topo podem devolver -nan(ind) (divisao por zero): vira BLANK.
