Barra Filtros Analitico 1 Agenda HTML =
// ============================================================================
// BARRA DE FILTROS - PAGINA "ANALÍTICO 1 AGENDA"  (v1.1)
// Fica POR BAIXO das segmentacoes nativas e de um botao nativo. O visual HTML
// nao filtra nem navega: ele so desenha a faixa escura, o titulo, uma "caixa"
// com rotulo para cada filtro, a data da ultima atualizacao e o botao
// VOLTAR. Mesma barra da pagina do painel, com o filtro de SITUACAO no lugar
// do mes (a carteira e de hoje) e a dica de como baixar.
//
// COMO MONTAR (barra com 1032 x 62 px, a mesma largura do painel)
//   1. Visual HTML com esta medida: altura 62 px, sem titulo, fundo transparente.
//   2. Uma segmentacao nativa (estilo Lista suspensa) em cima de cada caixa:
//        160 x 26 px, todas a 24 px do topo da barra. A borda esquerda de cada
//        uma fica, contando da borda DIREITA da barra: AT 432 px, Cor 610 px,
//        Situacao 788 px.
//   3. Formato de cada segmentacao: Cabecalho da segmentacao DESLIGADO;
//        Valores com fonte Segoe UI 10, cor BRANCA, Tela de fundo desligada;
//        em Geral > Efeitos: Tela de fundo, Borda e Sombra DESLIGADAS.
//   4. Botao nativo (Inserir > Botoes > Em branco) em cima do VOLTAR:
//        116 x 32 px, a 132 px da borda direita da barra e 15 px do topo.
//        Acao > Tipo: Voltar (ou Navegacao de pagina > EFETIVIDADE 1 AGENDA).
//        Preenchimento 100% transparente; em Estilo > "Ao passar o mouse",
//        preenchimento branco com 85% de transparencia.
//   5. Traga as segmentacoes e o botao para a frente.
//
// Para baixar: ... da tabela > Exportar dados > CSV (o Power BI nao tem acao
// de botao que baixe um arquivo).
//
// Para mudar os rotulos ou a quantidade de caixas, edite a tabela _filtros
// (cada caixa ocupa 178 px, contando o intervalo).
// PALETA: troque so a palavra em _tema ->  "GRAFITE" | "OCEANO" | "AURORA"
// O bloco de CSS fica dentro da div principal e nao tem o sinal de menor.
// ============================================================================
VAR _tema = "GRAFITE"
VAR _tokens =
    SWITCH(
        _tema,
        "AURORA",
            "--d0:#140A2E;--d1:#22114F;--a1:#F472B6;--g1:rgba(244,114,182,.24);--g2:rgba(167,139,250,.30)",
        "OCEANO",
            "--d0:#050E26;--d1:#0A1A42;--a1:#22D3EE;--g1:rgba(34,211,238,.22);--g2:rgba(59,130,246,.30)",
        "--d0:#0C1016;--d1:#161C26;--a1:#7DD3FC;--g1:rgba(125,211,252,.14);--g2:rgba(148,163,184,.20)"
    )

VAR _titulo = "Analítico · 1ª agenda"
VAR _subtitulo = "Perus / Butantã"
// Linha de dica abaixo do titulo (deixe "" para nao mostrar).
VAR _dica = "CSV: <b>⋯</b> da tabela › <b>Exportar dados</b>"
// Botao da direita: "baixar" (BAIXAR CSV) | "voltar" (VOLTAR) | "" (sem botao)
VAR _botao = "voltar"

// Ultima atualizacao (deixe _rotAtualizacao = "" para nao mostrar o bloco).
VAR _rotAtualizacao = "Atualização Zeus"
// <<< MODELO: data (ou data e hora) da ultima carga. Se tiver uma coluna com a
//     hora da carga da Zeus, troque aqui. Sem hora, mostra so a data.
VAR _atualizacao =
    CALCULATE( MAX( 'BACKLOG_INSTALACAO'[DATA_CARGA] ), REMOVEFILTERS() )
VAR _fuso = -3

// Um rotulo por segmentacao, da esquerda para a direita.
VAR _filtros =
    DATATABLE(
        "Rotulo", STRING,
        "Ordem", INTEGER,
        {
            { "Situação", 1 },
            { "Cor", 2 },
            { "AT", 3 }
        }
    )
VAR _caixas =
    CONCATENATEX(
        _filtros,
        "<div class='fx'><small>" & [Rotulo] & "</small><i></i></div>",
        "",
        [Ordem], ASC
    )

// verde = carga de hoje | amarelo = carga de outro dia
VAR _hojeN = INT( UTCNOW() + _fuso / 24 )
VAR _temHora = NOT ISBLANK( _atualizacao ) && _atualizacao - INT( _atualizacao ) > 0
VAR _blocoAtualizacao =
    IF(
        _rotAtualizacao = "",
        "",
        "<div class='up'><small>" & _rotAtualizacao & "</small>" &
            "<b class='" & IF( NOT ISBLANK( _atualizacao ) && INT( _atualizacao ) = _hojeN, "ok", "late" ) & "'><i></i>" &
                IF( ISBLANK( _atualizacao ), "sem data", FORMAT( _atualizacao, IF( _temHora, "dd/MM/yyyy hh:nn", "dd/MM/yyyy" ), "pt-BR" ) ) &
            "</b></div>"
    )
VAR _iconeBaixar = "<svg viewBox='0 0 24 24' aria-hidden='true'><path d='M12 4v11M7 10.5l5 5 5-5M5 19.5h14'></path></svg>"
VAR _iconeVoltar = "<svg viewBox='0 0 24 24' aria-hidden='true'><path d='M19 12H6M11 6.5 5.5 12l5.5 5.5'></path></svg>"
VAR _blocoBotao =
    SWITCH(
        _botao,
        "baixar", "<div class='bt'>" & _iconeBaixar & "<span>Baixar CSV</span></div>",
        "voltar", "<div class='bt'>" & _iconeVoltar & "<span>Voltar</span></div>",
        ""
    )

VAR _css =
"<style>
html,body{margin:0;padding:0;width:100%;height:100vh;overflow:hidden;background:transparent}
.bf,.bf *{box-sizing:border-box}
.bf i,.bf em{font-style:normal}
.bf{" & _tokens & ";position:relative;width:100%;height:100vh;padding:0 4px}
.bx{position:relative;width:100%;height:100%;display:flex;align-items:center;gap:10px;padding:0 12px 0 18px;overflow:hidden;border-radius:16px;color:#FFFFFF;
font-family:'Segoe UI',system-ui,-apple-system,Arial,sans-serif;text-align:left;-webkit-font-smoothing:antialiased;
background:radial-gradient(60% 200% at 0% 0%,var(--g2),transparent 62%),radial-gradient(40% 200% at 100% 100%,var(--g1),transparent 60%),linear-gradient(115deg,var(--d1),var(--d0));
box-shadow:0 8px 20px rgba(5,14,38,.20),inset 0 0 0 1px rgba(255,255,255,.08)}
.tt{flex:1 1 auto;min-width:0}
.tt b{display:block;overflow:hidden;text-overflow:ellipsis;font-size:16px;font-weight:700;letter-spacing:-.01em;line-height:20px;white-space:nowrap}
.tt small{display:block;margin-top:0;overflow:hidden;text-overflow:ellipsis;font-size:8.5px;font-weight:700;letter-spacing:.14em;line-height:13px;text-transform:uppercase;color:var(--a1);white-space:nowrap}
.tt em{display:block;margin-top:3px;overflow:hidden;text-overflow:ellipsis;font-size:9.5px;line-height:12px;color:rgba(255,255,255,.66);white-space:nowrap}
.tt em b{display:inline;font-size:9.5px;line-height:12px;letter-spacing:0;color:#FFFFFF}
/* GEOMETRIA: caixa de 168 px, intervalo de 10 px; rotulo com 15 px e area da segmentacao com 28 px */
.fx{flex:none;width:168px;height:46px}
.fx small{display:block;height:15px;padding-left:2px;font-size:8px;font-weight:700;letter-spacing:.12em;line-height:13px;text-transform:uppercase;color:rgba(255,255,255,.62)}
.fx i{display:block;height:28px;border-radius:9px;background:rgba(255,255,255,.08);box-shadow:inset 0 0 0 1px rgba(255,255,255,.14)}
/* GEOMETRIA: atualizacao e botao com 116 px cada; botao com 32 px de altura */
.up{flex:none;width:116px;min-width:0;padding-left:10px;border-left:1px solid rgba(255,255,255,.12)}
.up small{display:block;overflow:hidden;text-overflow:ellipsis;font-size:7.5px;font-weight:700;letter-spacing:.1em;line-height:11px;text-transform:uppercase;color:rgba(255,255,255,.58);white-space:nowrap}
.up b{display:flex;align-items:center;gap:6px;margin-top:3px;overflow:hidden;font-size:11px;font-weight:700;line-height:14px;white-space:nowrap}
.up b i{flex:none;width:7px;height:7px;border-radius:50%;background:#34D399;box-shadow:0 0 0 3px rgba(52,211,153,.18)}
.up b.late i{background:#FBBF24;box-shadow:0 0 0 3px rgba(251,191,36,.18)}
.bt{flex:none;width:116px;height:32px;display:flex;align-items:center;justify-content:center;gap:7px;border-radius:10px;background:rgba(255,255,255,.06);box-shadow:inset 0 0 0 1px rgba(255,255,255,.26)}
.bt svg{flex:none;width:15px;height:15px;fill:none;stroke:var(--a1);stroke-width:2.2;stroke-linecap:round;stroke-linejoin:round}
.bt span{font-size:9.5px;font-weight:700;letter-spacing:.09em;text-transform:uppercase;white-space:nowrap}
@media (max-width:760px){.tt{display:none}.bx{justify-content:flex-end}}
</style>"

VAR _html =
"<div class='bf'>" &
    _css &
    "<div class='bx'>" &
        "<div class='tt'><b>" & _titulo & "</b><small>" & _subtitulo & "</small>" & IF( _dica = "", "", "<em>" & _dica & "</em>" ) & "</div>" &
        _caixas &
        _blocoAtualizacao &
        _blocoBotao &
    "</div>" &
"</div>"

RETURN
    _html
