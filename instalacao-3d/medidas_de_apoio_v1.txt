// Sugestao: guardar na dMedidas, na pasta "Medidas INSTALA3D".
//
// As duas tabelas precisam estar relacionadas com a dCalendario (pela data) e

// ----------------------------------------------------------------------------
// 1) NUMERADOR: instalacoes dentro do prazo de 3 dias
// ----------------------------------------------------------------------------
INSTALACAO_3D NO PRAZO =
SUM( 'fINSTALACAO + ME_3D'[num_Total] )


// ----------------------------------------------------------------------------
// 2) DENOMINADOR: total de instalacoes
// ----------------------------------------------------------------------------
INSTALACAO_3D TOTAL =
SUM( 'fINSTALACAO + ME_3D_TOTAL'[num_Total] )


// ----------------------------------------------------------------------------
// 3) INDICADOR - a sua medida, reescrita em cima das duas acima (OPCIONAL)
//    Da o mesmo numero de hoje. A diferenca e o DIVIDE: quando o total e zero
//    (um dia ou uma AT sem instalacao) ele devolve vazio, e a barra "/" devolve
//    erro ou "NaN" na tela.
// ----------------------------------------------------------------------------
% INSTALACAO_3D =
DIVIDE( [INSTALACAO_3D NO PRAZO], [INSTALACAO_3D TOTAL] )
