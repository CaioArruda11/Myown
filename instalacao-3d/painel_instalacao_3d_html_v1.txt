Painel Instalacao 3D HTML =

// ============================================================================
// PAINEL "INSTALACAO EM ATE 3 DIAS"  (v1.2 - ligado ao modelo real)
// Um unico visual HTML Content ocupando o corpo da pagina. Fundo TRANSPARENTE.
// Mesma linguagem visual do BSC v5 e do Ranking v2.
//
// v1.2: CSS a prova de filtro (ver COMPATIBILIDADE) e coluna de agenda ligada.
// v1.1: usa as tabelas do arquivo novo
//   - indicador: 'fINSTALACAO + ME_3D' (no prazo) e 'fINSTALACAO + ME_3D_TOTAL'
//     (total), pelas medidas [INSTALACAO_3D NO PRAZO] e [INSTALACAO_3D TOTAL];
//   - o eixo diario mostra os dias que existem na dCalendario; o prazo do
//     lado D-0 pode ser contado em dias corridos (padrao) ou uteis;
//   - lado D-0 le a 'ETA DIRECT D-0'. FALTA informar a coluna com a data de
//     abertura da ordem (ver _pendModelo); ate la o lado direito avisa o que
//     falta em vez de mostrar numeros.
//
// PALETA: troque so a palavra em _tema ->  "GRAFITE" | "OCEANO" | "AURORA"
//
// O PAINEL TEM DUAS METADES
//   ESQUERDA - muda com o filtro de MES e com os botoes Mes | Semana | Dia:
//     1. faixa escura com o resultado do mes do filtro (meta, distancia da
//        meta, volumes, variacao contra o mes anterior) e as 4 CORES;
//     2. CURVA do indicador: 12 meses, 12 semanas ou os dias do mes. Os tres
//        graficos ja saem prontos e o clique so troca o que aparece;
//     3. ABERTURA por COR e por AT alinhada com a curva: cada linha e uma cor
//        ou uma AT e cada coluna e o mesmo periodo do grafico (mapa de calor).
//   DIREITA - FIXA NO DIA DE HOJE (D-0), ignora o filtro de mes:
//     4. carteira pendente na carga de hoje: ja perdidas (passaram do prazo),
//        vencem hoje, vencem amanha e no prazo;
//     5. a mesma conta por COR e por AT;
//     6. lista das ORDENS, com abas Perdidas | Vencem hoje | Amanha.
//   As duas metades respeitam os filtros de COR e de AT.
//
// O QUE AJUSTAR
//   - bloco 1 (CONFIGURACAO): meta, prazo (dias corridos ou uteis), fuso,
//     quantos meses/semanas;
//   - VARs marcadas com "<<< MODELO": sao as UNICAS que leem tabelas, colunas
//     e medidas. Todo o resto usa so o que essas VARs devolvem.
//   - medidas usadas: [INSTALACAO_3D NO PRAZO] e [INSTALACAO_3D TOTAL]
//     (arquivo medidas_de_apoio_v1.txt).
//
// COMPATIBILIDADE COM O VISUAL HTML (nao executa JavaScript)
//   - so HTML + CSS; botoes com <details> + :has(), como nas paginas IFI/IRR;
//   - o bloco <style> fica dentro da div principal e o CSS nao tem o sinal de
//     menor (v1.2): sem isso, visuais que filtram o HTML descartam o CSS todo;
//   - atributo style so carrega variaveis CSS e fica em um <div class='v'>
//     neutro; nenhum nome entra em atributo (title), so em texto;
//   - 100vh de proposito: o visual envolve o conteudo em um wrapper sem altura;
//   - nao usar aspas duplas dentro do CSS/HTML (quebram a string DAX).
// ============================================================================

// ------------------------------------------------------------------- PALETA
VAR _tema = "GRAFITE"

// d0/d1/d2 = fundos escuros | a1/a2 = cores de destaque | g1/g2 = brilhos
// pri/pri-b = destaque sobre fundo claro
VAR _tokens =
    SWITCH(
        _tema,
        "AURORA",
            "--d0:#140A2E;--d1:#22114F;--d2:#43208A;--a1:#F472B6;--a2:#A78BFA;--g1:rgba(244,114,182,.24);--g2:rgba(167,139,250,.30);--pri:#7C3AED;--pri-b:#F3EDFF",
        "OCEANO",
            "--d0:#050E26;--d1:#0A1A42;--d2:#123A7C;--a1:#22D3EE;--a2:#60A5FA;--g1:rgba(34,211,238,.22);--g2:rgba(59,130,246,.30);--pri:#2563EB;--pri-b:#E9F0FF",
        "--d0:#0C1016;--d1:#161C26;--d2:#2B3442;--a1:#7DD3FC;--a2:#CBD5E1;--g1:rgba(125,211,252,.14);--g2:rgba(148,163,184,.20);--pri:#334155;--pri-b:#EEF2F6"
    )

// ============================================================================
// 1. CONFIGURACAO  (AJUSTE AQUI)
// ============================================================================
VAR _loc = "pt-BR"
VAR _meta = 0.81              // meta do indicador (81%)
VAR _prazoDias = 3            // prazo, em dias de peso (ver abaixo), contado do dia seguinte a criacao da ordem
// Quanto cada dia consome do prazo (o dia da criacao da ordem nao conta):
// dia util = 1, sabado = 0,75, domingo = 0,25. A equipe tem _prazoDias desses.
VAR _pesoUtil = 1
VAR _pesoSab = 0.75
VAR _pesoDom = 0.25
VAR _fuso = -3                // fuso de Brasilia (UTC-3)
VAR _qtdMeses = 12            // meses na curva mensal
VAR _qtdSemanas = 12          // semanas na curva semanal (segunda a domingo)
VAR _visaoInicial = "Mês"     // curva que abre primeiro: "Mês" | "Semana" | "Dia"
VAR _maxLinhas = 60           // ordens listadas em cada aba (as mais urgentes)
// Valor da coluna 'ETA DIRECT D-0'[Indicador Instalação 3D] que significa que a
// ordem AINDA conta dentro do indicador. Qualquer outro valor = ja perdida.
VAR _valorDentro = "DENTRO DO INDICADOR"

// Cores das areas: nome (como esta na coluna de cor), ordem na tela e classe.
VAR _coresCfg =
    DATATABLE(
        "CorCfg", STRING,
        "OrdCor", INTEGER,
        "ClsCor", STRING,
        {
            { "OURO", 1, "c-ouro" },
            { "PRATA 1", 2, "c-prata1" },
            { "PRATA 2", 3, "c-prata2" },
            { "BRONZE", 4, "c-bronze" }
        }
    )

// ============================================================================
// 2. DATAS DE REFERENCIA
// ============================================================================
// "Hoje" no fuso informado (UTCNOW e igual no Desktop e no Service).
VAR _agora = UTCNOW() + _fuso / 24
VAR _hoje = DATE( YEAR( _agora ), MONTH( _agora ), DAY( _agora ) )
VAR _hojeN = INT( _hoje )

// <<< MODELO: ultima data do filtro de periodo da pagina (segmentacao de Mes).
VAR _fimSel = MAX( dCalendario[Data] )

// Mes de referencia = mes do filtro. Sem filtro, ou com mes futuro: mes atual.
VAR _ref = IF( ISBLANK( _fimSel ) || _fimSel > _hoje, _hoje, _fimSel )
VAR _mesIni = DATE( YEAR( _ref ), MONTH( _ref ), 1 )
VAR _mesFim = EOMONTH( _ref, 0 )
VAR _mesAntIni = EDATE( _mesIni, -1 )
VAR _mesIni0 = EDATE( _mesIni, 1 - _qtdMeses )               // 1o mes da curva mensal
VAR _semRef = _ref - WEEKDAY( _ref, 2 ) + 1                   // segunda-feira da semana de referencia
VAR _semIni0 = _semRef - 7 * ( _qtdSemanas - 1 )              // 1a semana da curva semanal
VAR _janIni = MIN( _mesIni0, _semIni0 )
VAR _janFim = MAX( _mesFim, _semRef + 6 )
// dias do calendario: a janela da curva + os ultimos 90 dias (prazo do lado D-0)
VAR _calIni = MIN( _janIni, _hoje - 90 )
VAR _calFim = MAX( _janFim, _hoje + 30 )

// ============================================================================
// 3. MODELO  (unicas VARs, junto com _fimSel, que leem o modelo)
// ============================================================================
// <<< MODELO: uma linha por AT x dia com as duas medidas de volume.
//     - ignora o filtro de mes (a curva sempre mostra a janela inteira);
//     - ignora o filtro de cor/AT (para comparar a selecao com o Geral e com
//       as cores); o filtro e aplicado depois, em memoria, com _atsCtx.
//     As duas tabelas fINSTALACAO precisam estar relacionadas com a
//     dCalendario (data) e com a dAT (AT), como ja estao para a sua medida.
VAR _gradeModelo =
    CALCULATETABLE(
        ADDCOLUMNS(
            CROSSJOIN(
                SUMMARIZE( dAT, dAT[AREA ], dAT[COR_AREA ] ),
                VALUES( dCalendario[Data] )
            ),
            "@n", [INSTALACAO_3D NO PRAZO],
            "@d", [INSTALACAO_3D TOTAL]
        ),
        REMOVEFILTERS( dAT ),
        REMOVEFILTERS( dCalendario ),
        dCalendario[Data] >= _janIni && dCalendario[Data] <= _janFim
    )

// <<< MODELO: ATs que estao no filtro da pagina (cor e/ou AT selecionadas).
VAR _atsCtx = VALUES( dAT[AREA ] )

// <<< MODELO: os dias que existem na dCalendario. Servem para montar o eixo
//     diario e para contar o prazo quando ele e em dias uteis.
VAR _calModelo =
    CALCULATETABLE(
        VALUES( dCalendario[Data] ),
        REMOVEFILTERS( dCalendario ),
        dCalendario[Data] >= _calIni && dCalendario[Data] <= _calFim
    )

// <<< MODELO: carga do dia = 'ETA DIRECT D-0'. IGNORA o filtro de mes.
//     O filtro de cor/AT e aplicado mais abaixo pela coluna AT, entao nao
//     precisa de relacionamento com a dAT (os nomes das ATs tem que ser os
//     mesmos da dAT[AREA ]).
//
//         "Ind"      = coluna calculada [Indicador Instalação 3D]: diz se a ordem
//                      ainda esta DENTRO DO INDICADOR (ver _valorDentro). E ela
//                      que separa as ja perdidas das demais.
//         "Abertura" = [Data Agendada para Execução]: e a data de CRIACAO da ordem
//                      (o prazo conta a partir do dia seguinte). Com ela o painel
//                      separa quem vence hoje e quem vence amanha.
//         "Agenda"   = nao usada (BLANK): a data agendada ja e a de criacao.
//     >>> Se a carga tiver reparos ou atividades ja concluidas, filtre aqui,
//         trocando a linha  'ETA DIRECT D-0',  por algo como:
//         FILTER( 'ETA DIRECT D-0', 'ETA DIRECT D-0'[coluna do tipo] = "Instalação"
//                                   && 'ETA DIRECT D-0'[coluna do status] <> "Concluída" ),
VAR _pendModelo =
    CALCULATETABLE(
        SELECTCOLUMNS(
            'ETA DIRECT D-0',
            "Ordem", "" & 'ETA DIRECT D-0'[Atividade],
            "AT", TRIM( 'ETA DIRECT D-0'[AT] ),
            "Ind", UPPER( TRIM( 'ETA DIRECT D-0'[Indicador Instalação 3D] ) ),
            "Abertura", 'ETA DIRECT D-0'[Data Agendada para Execução],
            "Agenda", BLANK()
        ),
        REMOVEFILTERS( dCalendario )
    )

// <<< MODELO (opcional): data/hora da ultima carga da 'ETA DIRECT D-0', para
//     avisar quando a carga de hoje ainda nao entrou. Exemplo:
//     CALCULATE( MAX( 'ETA DIRECT D-0'[coluna da data da carga] ), REMOVEFILTERS() )
VAR _dataCarga = BLANK()

// ============================================================================
// 4. GRADE AT x DIA  ->  AT / COR / GERAL / SELECAO  x  MES / SEMANA / DIA
// Daqui para baixo nada mais consulta o modelo: e tudo soma em memoria, por
// isso os numeros fecham entre si (dia soma na semana, AT soma na cor...).
// ============================================================================
VAR _grade =
    SELECTCOLUMNS(
        FILTER( _gradeModelo, [@d] > 0 ),
        "AT", dAT[AREA ],
        "Cor", UPPER( TRIM( dAT[COR_AREA ] ) ),
        "Dia", dCalendario[Data],
        "N", COALESCE( [@n], 0 ),
        "D", [@d]
    )
VAR _gradeChaves =
    ADDCOLUMNS(
        _grade,
        "kM", DATE( YEAR( [Dia] ), MONTH( [Dia] ), 1 ),
        "kS", [Dia] - WEEKDAY( [Dia], 2 ) + 1
    )

// Uma linha por granularidade x AT x periodo. Gran: M = mes, S = semana, D = dia.
VAR _longAT =
    UNION(
        SELECTCOLUMNS(
            GROUPBY(
                FILTER( _gradeChaves, [kM] >= _mesIni0 && [kM] <= _mesIni ),
                [AT], [Cor], [kM],
                "sN", SUMX( CURRENTGROUP(), [N] ),
                "sD", SUMX( CURRENTGROUP(), [D] )
            ),
            "Gran", "M", "AT", [AT], "Cor", [Cor], "Ini", [kM], "N", [sN], "D", [sD]
        ),
        SELECTCOLUMNS(
            GROUPBY(
                FILTER( _gradeChaves, [kS] >= _semIni0 && [kS] <= _semRef ),
                [AT], [Cor], [kS],
                "sN", SUMX( CURRENTGROUP(), [N] ),
                "sD", SUMX( CURRENTGROUP(), [D] )
            ),
            "Gran", "S", "AT", [AT], "Cor", [Cor], "Ini", [kS], "N", [sN], "D", [sD]
        ),
        SELECTCOLUMNS(
            FILTER( _gradeChaves, [Dia] >= _mesIni && [Dia] <= _mesFim ),
            "Gran", "D", "AT", [AT], "Cor", [Cor], "Ini", [Dia], "N", [N], "D", [D]
        )
    )

// ---- ATs e cores: todas (com instalacao na janela) x as do filtro ----
VAR _atsTodas = GROUPBY( _grade, [AT], [Cor] )
VAR _atsVis = FILTER( _atsTodas, CONTAINSROW( _atsCtx, [AT] ) )
VAR _qtdTodas = COUNTROWS( _atsTodas ) + 0
VAR _qtdVis = COUNTROWS( _atsVis ) + 0
VAR _coresVis = GROUPBY( _atsVis, [Cor] )
VAR _qtdCoresVis = COUNTROWS( _coresVis ) + 0
VAR _corUnica = IF( _qtdCoresVis = 1, MAXX( _coresVis, [Cor] ) )
VAR _qtdDaCorUnica = COUNTROWS( FILTER( _atsTodas, [Cor] = _corUnica ) ) + 0

// Contexto do filtro: G = geral | C = uma cor inteira | A = uma AT | S = selecao
// de varias ATs | V = nada no filtro tem instalacao na janela.
VAR _ctx =
    SWITCH(
        TRUE(),
        _qtdVis = 0, "V",
        _qtdVis = _qtdTodas, "G",
        _qtdVis = 1, "A",
        _qtdCoresVis = 1 && _qtdVis = _qtdDaCorUnica, "C",
        "S"
    )
// Serie "principal" = a que vira a curva e o numero grande.
VAR _entPrincipal =
    SWITCH( _ctx, "G", "#G", "A", "A:" & MAXX( _atsVis, [AT] ), "C", "C:" & _corUnica, "#S" )
VAR _nomeCtx =
    SWITCH( _ctx, "G", "Geral", "A", "AT " & MAXX( _atsVis, [AT] ), "C", _corUnica, "V", "Sem dados", "Seleção" )
VAR _subCtx =
    SWITCH(
        _ctx,
        "A", _corUnica,
        "V", "nenhuma instalação no filtro",
        _qtdVis & IF( _qtdVis = 1, " AT", " ATs" )
    )
VAR _clsCtx = COALESCE( MAXX( FILTER( _coresCfg, [CorCfg] = _corUnica ), [ClsCor] ), "c-geral" )

// ---- series: Geral, selecao, cada cor (inteira) e cada AT do filtro ----
VAR _longNoFiltro = FILTER( _longAT, CONTAINSROW( _atsCtx, [AT] ) )
VAR _series =
    UNION(
        SELECTCOLUMNS(
            GROUPBY( _longAT, [Gran], [Ini], "sN", SUMX( CURRENTGROUP(), [N] ), "sD", SUMX( CURRENTGROUP(), [D] ) ),
            "Ent", "#G", "Gran", [Gran], "Ini", [Ini], "N", [sN], "D", [sD]
        ),
        SELECTCOLUMNS(
            GROUPBY( _longNoFiltro, [Gran], [Ini], "sN", SUMX( CURRENTGROUP(), [N] ), "sD", SUMX( CURRENTGROUP(), [D] ) ),
            "Ent", "#S", "Gran", [Gran], "Ini", [Ini], "N", [sN], "D", [sD]
        ),
        SELECTCOLUMNS(
            GROUPBY( _longAT, [Gran], [Cor], [Ini], "sN", SUMX( CURRENTGROUP(), [N] ), "sD", SUMX( CURRENTGROUP(), [D] ) ),
            "Ent", "C:" & [Cor], "Gran", [Gran], "Ini", [Ini], "N", [sN], "D", [sD]
        ),
        SELECTCOLUMNS(
            _longNoFiltro,
            "Ent", "A:" & [AT], "Gran", [Gran], "Ini", [Ini], "N", [N], "D", [D]
        )
    )

// ---- periodos de cada curva ----
// Rot = rotulo do eixo | RotM = rotulo medio (resumo) | RotL = rotulo longo (dica)
// _diasCal = dias que existem na dCalendario | _primeiroDia = inicio dos dados
VAR _diasCal = SELECTCOLUMNS( _calModelo, "Dia", dCalendario[Data] )
VAR _primeiroDia = MINX( _grade, [Dia] )
VAR _perM =
    SELECTCOLUMNS(
        ADDCOLUMNS( GENERATESERIES( 1, _qtdMeses ), "@i", EDATE( _mesIni, [Value] - _qtdMeses ) ),
        "Gran", "M",
        "Ord", [Value],
        "Ini", [@i],
        "Fim", EOMONTH( [@i], 0 ),
        "Rot", FORMAT( [@i], "MMM/yy", _loc ),
        "RotM", FORMAT( [@i], "MMM/yy", _loc ),
        "RotL", FORMAT( [@i], "MMMM/yyyy", _loc ),
        "Fds", 0
    )
VAR _perS =
    SELECTCOLUMNS(
        ADDCOLUMNS( GENERATESERIES( 1, _qtdSemanas ), "@i", _semIni0 + 7 * ( [Value] - 1 ) ),
        "Gran", "S",
        "Ord", [Value],
        "Ini", [@i],
        "Fim", [@i] + 6,
        // S + numero da semana no ano: o mesmo numero da coluna Semana do Ano da dCalendario
        "Rot", "S" & WEEKNUM( [@i], 2 ),
        "RotM", "S" & WEEKNUM( [@i], 2 ),
        "RotL", "Semana " & WEEKNUM( [@i], 2 ) & " · " & FORMAT( [@i], "dd/MM", _loc ) & " a " & FORMAT( [@i] + 6, "dd/MM", _loc ),
        "Fds", 0
    )
// Dia: os dias do mes que existem na dCalendario.
VAR _diasMes = FILTER( _diasCal, [Dia] >= _mesIni && [Dia] <= _mesFim )
VAR _perD =
    SELECTCOLUMNS(
        _diasMes,
        "Gran", "D",
        "Ord",
            VAR _d0 = [Dia]
            RETURN COUNTROWS( FILTER( _diasMes, [Dia] <= _d0 ) ),
        "Ini", [Dia],
        "Fim", [Dia],
        "Rot", FORMAT( [Dia], "dd", _loc ),
        "RotM", FORMAT( [Dia], "dd/MM", _loc ),
        "RotL", FORMAT( [Dia], "ddd, dd/MM", _loc ),
        "Fds", IF( WEEKDAY( [Dia], 2 ) >= 6, 1, 0 )
    )
// A curva comeca onde os dados comecam: periodos anteriores a primeira
// instalacao da base ficam de fora (base nova tem menos de 12 meses).
VAR _per =
    FILTER(
        UNION( _perM, _perS, _perD ),
        ISBLANK( _primeiroDia ) || [Fim] >= _primeiroDia
    )

// ---- serie principal em cada periodo ----
VAR _seriePrincipal = FILTER( _series, [Ent] = _entPrincipal )
VAR _serieP0 =
    ADDCOLUMNS(
        _per,
        "pN",
            VAR _gN = [Gran]
            VAR _iN = [Ini]
            RETURN SUMX( FILTER( _seriePrincipal, [Gran] = _gN && [Ini] = _iN ), [N] ),
        "pD",
            VAR _gD = [Gran]
            VAR _iD = [Ini]
            RETURN SUMX( FILTER( _seriePrincipal, [Gran] = _gD && [Ini] = _iD ), [D] )
    )
VAR _serieP = ADDCOLUMNS( _serieP0, "P", DIVIDE( [pN], [pD] ) )

// ---- escala de status: resultado / meta ----
// >= 100% da meta | 90-99% | 80-89% | < 80%   (mesma escala do BSC)
VAR _lim90 = _meta * 0.90
VAR _lim80 = _meta * 0.80

// ============================================================================
// 5. FAIXA DO PERIODO (mes do filtro)
// ============================================================================
VAR _regMes = FILTER( _seriePrincipal, [Gran] = "M" && [Ini] = _mesIni )
VAR _mesN = SUMX( _regMes, [N] )
VAR _mesD = SUMX( _regMes, [D] )
VAR _mesP = DIVIDE( _mesN, _mesD )
VAR _regAnt = FILTER( _seriePrincipal, [Gran] = "M" && [Ini] = _mesAntIni )
VAR _antP = DIVIDE( SUMX( _regAnt, [N] ), SUMX( _regAnt, [D] ) )
VAR _difAnt = IF( NOT ISBLANK( _mesP ) && NOT ISBLANK( _antP ), _mesP - _antP )
VAR _ultDia = MAXX( FILTER( _grade, [Dia] >= _mesIni && [Dia] <= _mesFim ), [Dia] )
VAR _mesEmAndamento = _mesFim >= _hoje
VAR _mesRot = FORMAT( _mesIni, "MMMM/yyyy", _loc )
VAR _tierMes =
    SWITCH( TRUE(), ISBLANK( _mesP ), "tN", _mesP >= _meta, "tG", _mesP >= _lim90, "tY", _mesP >= _lim80, "tO", "tR" )
VAR _chipMes =
    SWITCH(
        TRUE(),
        ISBLANK( _mesP ), "sem dados",
        _mesP >= _meta, "▲ " & FORMAT( ( _mesP - _meta ) * 100, "0.0", _loc ) & " p.p. acima da meta",
        "▼ " & FORMAT( ( _meta - _mesP ) * 100, "0.0", _loc ) & " p.p. abaixo da meta"
    )
// Quantas instalacoes no prazo sobraram (+) ou faltaram (-) para a meta do mes.
VAR _saldoMeta = IF( ISBLANK( _mesD ), BLANK(), _mesN - _meta * _mesD )

VAR _heroResultado =
    "<div class='hz hk " & _tierMes & "'>" &
        "<span class='kk'>Instalação em até " & _prazoDias & " dias</span>" &
        "<div class='kv'><b>" & IF( ISBLANK( _mesP ), "—", FORMAT( _mesP, "0.0%", _loc ) ) & "</b><span class='chip'>" & _chipMes & "</span></div>" &
        "<span class='ks km'><b>" & _mesRot & "</b>" &
            IF( _mesEmAndamento && NOT ISBLANK( _ultDia ), " · parcial até " & FORMAT( _ultDia, "dd/MM", _loc ), "" ) &
            " · meta ≥ " & FORMAT( _meta, "0.0%", _loc ) & "</span>" &
        "<span class='ks'>" &
            SWITCH(
                TRUE(),
                ISBLANK( _saldoMeta ), "sem instalações concluídas no mês",
                _saldoMeta >= 0, "folga de <b>" & FORMAT( ROUNDDOWN( _saldoMeta, 0 ), "#,0", _loc ) & "</b> instalações no prazo sobre a meta",
                "faltam <b>" & FORMAT( ROUNDUP( - _saldoMeta, 0 ), "#,0", _loc ) & "</b> instalações no prazo para a meta"
            ) & "</span>" &
    "</div>"

VAR _heroVolumes =
    "<div class='hz hv'>" &
        "<span class='kc " & _clsCtx & "'><i class='cd'></i>" & _nomeCtx & "<small>" & _subCtx & "</small></span>" &
        "<div class='kp'>" &
            "<div><small>Instaladas</small><b>" & IF( ISBLANK( _mesD ), "—", FORMAT( _mesD, "#,0", _loc ) ) & "</b></div>" &
            "<div class='tG'><small><i></i>No prazo</small><b>" & IF( ISBLANK( _mesD ), "—", FORMAT( _mesN, "#,0", _loc ) ) & "</b></div>" &
            "<div class='tR'><small><i></i>Fora do prazo</small><b>" & IF( ISBLANK( _mesD ), "—", FORMAT( _mesD - _mesN, "#,0", _loc ) ) & "</b></div>" &
        "</div>" &
        "<span class='kd'>" &
            IF(
                ISBLANK( _difAnt ),
                "sem mês anterior para comparar",

// Medidas de referencia do grafico (px) usadas so para engrossar o traco nos
// trechos inclinados; nao precisam ser exatas.
VAR _refW = 600
VAR _refH = 130
VAR _traco = 1.1

VAR _views =
    CONCATENATEX(
        _grans,
        VAR _g = [G]
        VAR _sufV = [SufV]
        VAR _rotAc = [RotAc]
        VAR _dicaAc = [DicaAc]
        VAR _ordG = [OrdG]
        VAR _dG = FILTER( _serieP, [Gran] = _g )
        VAR _nPer = COUNTROWS( _dG ) + 0
        VAR _ordMax = MAXX( _dG, [Ord] )
        VAR _ordMeio = ( MINX( _dG, [Ord] ) + _ordMax ) / 2
        VAR _comV = FILTER( _dG, NOT ISBLANK( [P] ) )
        VAR _minV = MINX( _comV, [P] )
        VAR _maxV = MAXX( _comV, [P] )

        // -- eixo Y: nao comeca no zero (e uma curva, nao barra); passo redondo
        VAR _lo = MIN( COALESCE( _minV, _meta ), _meta ) - 0.01
        VAR _hi = MAX( COALESCE( _maxV, _meta ), _meta ) + 0.01
        VAR _amp = _hi - _lo
        VAR _passo = SWITCH( TRUE(), _amp <= 0.10, 0.02, _amp <= 0.25, 0.05, _amp <= 0.50, 0.10, 0.20 )
        VAR _yMin = MAX( 0, ROUNDDOWN( ROUND( _lo / _passo, 6 ), 0 ) * _passo )
        VAR _yMax = MIN( 1, ROUNDUP( ROUND( _hi / _passo, 6 ), 0 ) * _passo )
        VAR _nPassos = ROUND( ( _yMax - _yMin ) / _passo, 0 )
        VAR _linhasGrade =
            CONCATENATEX(
                GENERATESERIES( 0, _nPassos ),
                VAR _valG = _yMin + [Value] * _passo
                RETURN
                    "<div class='v' style='--g:" & FORMAT( ( _yMax - _valG ) / ( _yMax - _yMin ) * 100, "0.00", "en-US" ) & "%'>" &
                    "<i class='gl'><b>" & FORMAT( _valG, "0%", _loc ) & "</b></i></div>",
                "",
                [Value], ASC
            )
        VAR _yMeta = ( _yMax - _meta ) / ( _yMax - _yMin ) * 100
        VAR _posLinhaMeta = FORMAT( _yMeta, "0.00", "en-US" )

        // -- pontos: Y = distancia do topo, em % da altura
        VAR _dY =
            ADDCOLUMNS( _dG, "Y", IF( ISBLANK( [P] ), BLANK(), ( _yMax - [P] ) / ( _yMax - _yMin ) * 100 ) )
        VAR _ultOrd = MAXX( _comV, [Ord] )
        // O unico rotulo fixo e o do ultimo ponto, a direita dele (o melhor e o pior
        // estao no resumo do cabecalho; os demais na dica e na linha da tabela).
        // Se ele cair na altura do rotulo da meta, o da meta sobe ou desce.
        VAR _yUlt = MAXX( FILTER( _dY, [Ord] = _ultOrd ), [Y] )
        VAR _clsMeta =
            IF(
                _ultOrd = _ordMax && NOT ISBLANK( _yUlt ) && ABS( _yUlt - _yMeta ) < 16,
                IF( _yUlt >= _yMeta, " up", " dw" ),
                ""
            )
        VAR _meiaCol = _refW / MAX( _nPer, 1 ) / 2
        VAR _pontos =
            CONCATENATEX(
                _dY,
                VAR _o = [Ord]
                VAR _y = [Y]
                VAR _pv = [P]
                VAR _futuro = [Ini] > _hoje
                VAR _atual = [Ini] <= _hoje && [Fim] >= _hoje
                VAR _yAnt = MAXX( FILTER( _dY, [Ord] = _o - 1 ), [Y] )
                VAR _yPro = MAXX( FILTER( _dY, [Ord] = _o + 1 ), [Y] )
                // o traco vai do meio do caminho ate o ponto anterior ao meio do caminho ate o proximo
                VAR _ya = IF( ISBLANK( _yAnt ), _y, ( _y + _yAnt ) / 2 )
                VAR _yc = IF( ISBLANK( _yPro ), _y, ( _y + _yPro ) / 2 )
                VAR _ep = _traco * MIN( 3.5, SQRT( 1 + POWER( ( _ya - _y ) / 100 * _refH / _meiaCol, 2 ) ) )
                VAR _eq = _traco * MIN( 3.5, SQRT( 1 + POWER( ( _yc - _y ) / 100 * _refH / _meiaCol, 2 ) ) )
                VAR _tierP = SWITCH( TRUE(), _pv >= _meta, "tG", _pv >= _lim90, "tY", _pv >= _lim80, "tO", "tR" )
                VAR _dica =
                    "<span class='tip'><b>" & [RotL] & IF( _atual, " · em andamento", "" ) & "</b>" &
                        "<span><i></i>No prazo<em>" & FORMAT( _pv, "0.0%", _loc ) & "</em></span>" &
                        "<span>vs. meta<em class='" & IF( _pv >= _meta, "up", "dn" ) & "'>" & FORMAT( ( _pv - _meta ) * 100, "+0.0;-0.0;0.0", _loc ) & " p.p.</em></span>" &
                        "<span>Instaladas<em>" & FORMAT( [pD], "#,0", _loc ) & "</em></span>" &
                        "<span>No prazo<em>" & FORMAT( [pN], "#,0", _loc ) & "</em></span>" &
                        "<span>Fora do prazo<em>" & FORMAT( [pD] - [pN], "#,0", _loc ) & "</em></span>" &
                    "</span>"
                RETURN
                    IF(
                        ISBLANK( _pv ),
                        "<div class='pt nd" & IF( _futuro, " fu", "" ) & IF( _atual, " hj", "" ) & IF( _o > _ordMeio, " tr", "" ) & "'>" &
                            IF( _futuro, "", "<span class='tip'><b>" & [RotL] & "</b><span>Sem instalações concluídas</span></span>" ) &
                        "</div>",
                        "<div class='v' style='--y:" & FORMAT( _y, "0.00", "en-US" ) & "%;--a:" & FORMAT( _ya, "0.00", "en-US" ) & "%;--c:" & FORMAT( _yc, "0.00", "en-US" ) &
                            "%;--l:" & IF( ISBLANK( _yAnt ), "50%", "0%" ) & ";--r:" & IF( ISBLANK( _yPro ), "50%", "100%" ) &
                            ";--p:" & FORMAT( _ep, "0.00", "en-US" ) & "px;--q:" & FORMAT( _eq, "0.00", "en-US" ) & "px'>" &
                        "<div class='pt " & _tierP & IF( _atual, " hj", "" ) & IF( _o > _ordMeio, " tr", "" ) & "'>" &
                            "<i class='ln'></i><u class='dot'></u>" &
                            IF( _o = _ultOrd, "<em class='lb'>" & FORMAT( _pv, "0.0%", _loc ) & "</em>", "" ) &
                            _dica &
                        "</div></div>"
                    ),
                "",
                [Ord], ASC
            )

        // -- eixo X (tambem e o cabecalho das colunas da abertura)
        VAR _eixoX =
            CONCATENATEX(
                _dG,
                "<span class='xl" & IF( [Fds] = 1, " we", "" ) & IF( [Ini] <= _hoje && [Fim] >= _hoje, " hj", "" ) & "'>" & [Rot] & "</span>",
                "",
                [Ord], ASC
            )

        // -- abertura: uma linha por Geral / selecao / cor / AT, colunas = periodos
        VAR _linhasAbertura =
            CONCATENATEX(
                _linhas,
                VAR _e = [Ent]
                VAR _serieE = FILTER( _series, [Ent] = _e && [Gran] = _g )
                VAR _acN = SUMX( _serieE, [N] )
                VAR _acD = SUMX( _serieE, [D] )
                VAR _acP = DIVIDE( _acN, _acD )
                VAR _tierAc =
                    SWITCH( TRUE(), ISBLANK( _acP ), "tN", _acP >= _meta, "tG", _acP >= _lim90, "tY", _acP >= _lim80, "tO", "tR" )
                VAR _celulas =
                    CONCATENATEX(
                        _dG,
                        VAR _iC = [Ini]
                        VAR _regC = FILTER( _serieE, [Ini] = _iC )
                        VAR _dC = SUMX( _regC, [D] )
                        VAR _nC = SUMX( _regC, [N] )
                        VAR _pC = DIVIDE( _nC, _dC )
                        RETURN
                            IF(
                                ISBLANK( _pC ),
                                "<i class='c z'>" & IF( _iC > _hoje, "", "·" ) & "</i>",
                                "<i class='c " & SWITCH( TRUE(), _pC >= _meta, "tG", _pC >= _lim90, "tY", _pC >= _lim80, "tO", "tR" ) &
                                    "' title='" & [RotL] & " · " & FORMAT( _pC, "0.0%", _loc ) & " · " & FORMAT( _nC, "#,0", _loc ) & " de " & FORMAT( _dC, "#,0", _loc ) & " no prazo'>" &
                                    IF( _g = "D", FORMAT( _pC * 100, "0", _loc ), FORMAT( _pC * 100, "0.0", _loc ) ) & "</i>"
                            ),
                        "",
                        [Ord], ASC
                    )
                RETURN
                    "<div class='rw r-" & LOWER( [Tipo] ) & IF( [ClsL] = "", "", " " & [ClsL] ) & IF( _e = _entPrincipal, " pr", "" ) & "'>" &
                        "<span class='rl'>" & IF( [Tipo] = "C" || [Tipo] = "A", "<i class='cd'></i>", "" ) & "<b>" & [Nome] & "</b>" &
                            IF( [Sub] = "", "", "<small>" & [Sub] & "</small>" ) & "</span>" &
                        "<div class='cs'>" & _celulas & "</div>" &
                        "<b class='ac " & _tierAc & "'>" & IF( ISBLANK( _acP ), "—", FORMAT( _acP, "0.0%", _loc ) ) & "</b>" &
                    "</div>",
                "",
                [OrdGrp], ASC,
                [OrdTipo], ASC,
                [PM], DESC,
                [Nome], ASC
            )

        // -- volume da serie principal, nas mesmas colunas
        VAR _volTotal = SUMX( _dG, [pD] )
        VAR _linhaVolume =
            "<div class='rw r-q'>" &
                "<span class='rl'><b>Instaladas</b></span>" &
                "<div class='cs'>" &
                    CONCATENATEX(
                        _dG,
                        VAR _qv = [pD]
                        RETURN
                            "<i class='c q'>" &
                                SWITCH(
                                    TRUE(),
                                    ISBLANK( _qv ), "",
                                    _qv >= 10000, FORMAT( _qv / 1000, "0", _loc ) & "k",
                                    _qv >= 1000, FORMAT( _qv / 1000, "0.0", _loc ) & "k",
                                    FORMAT( _qv, "0", _loc )
                                ) & "</i>",
                        "",
                        [Ord], ASC
                    ) &
                "</div>" &
                "<b class='ac q'>" &
                    SWITCH(
                        TRUE(),
                        ISBLANK( _volTotal ), "—",
                        _volTotal >= 10000, FORMAT( _volTotal / 1000, "0.0", _loc ) & "k",
                        FORMAT( _volTotal, "#,0", _loc )
                    ) & "</b>" &
            "</div>"
        RETURN
            "<div class='v' style='--m:" & _posLinhaMeta & "%'>" &
            "<div class='view v-" & _sufV & IF( _ordG = _gA, " ini", "" ) & IF( _g = "D", " dense", "" ) & "'>" &
                "<div class='ch'>" &
                    "<div class='lg'>" &
                        "<span class='tG'><i></i>na meta</span>" &
                        "<span class='tY'><i></i>até 10% abaixo</span>" &
                        "<span class='tO'><i></i>até 20% abaixo</span>" &
                        "<span class='tR'><i></i>mais de 20%</span>" &
                    "</div>" &
                    "<div class='plot'>" & _linhasGrade & "<i class='ml" & _clsMeta & "'><b>meta " & FORMAT( _meta, "0%", _loc ) & "</b></i>" & _pontos & "</div>" &
                "</div>" &
                "<div class='xr'><span class='xk'>% no prazo</span><div class='xs'>" & _eixoX & "</div><span class='xa' title='" & _dicaAc & "'>" & _rotAc & "</span></div>" &
                "<div class='mx'>" & _linhasAbertura & _linhaVolume & "</div>" &
            "</div></div>",
        "",
        [OrdG], ASC
    )

// ============================================================================
// 8. HOJE (D-0): ORDENS PENDENTES NA CARGA DO DIA ('ETA DIRECT D-0')
// Idade = dias passados desde a abertura (uteis ou corridos, conforme
// pesos por dia). O ultimo dia do prazo e quando a idade chega a _prazoDias.
//   P = ja perdida (idade maior que o prazo)   H = vence hoje (idade = prazo)
//   A = vence amanha (idade = prazo - 1)       N = no prazo
// ============================================================================
VAR _rotDias = "dias com peso (útil " & FORMAT( _pesoUtil, "0.##", _loc ) & " · sáb " & FORMAT( _pesoSab, "0.##", _loc ) & " · dom " & FORMAT( _pesoDom, "0.##", _loc ) & ")"
// Peso do dia anterior (para saber quem perdeu o prazo so hoje)
VAR _wdOntem = WEEKDAY( _hojeN - 1, 2 )
VAR _wOntem = IF( _wdOntem <= 5, _pesoUtil, IF( _wdOntem = 6, _pesoSab, _pesoDom ) )
VAR _wdAmanha = WEEKDAY( _hojeN + 1, 2 )
VAR _wAmanha = IF( _wdAmanha <= 5, _pesoUtil, IF( _wdAmanha = 6, _pesoSab, _pesoDom ) )

VAR _qCarga = COUNTROWS( _pendModelo ) + 0
VAR _qComAbertura = COUNTROWS( FILTER( _pendModelo, NOT ISBLANK( [Abertura] ) ) ) + 0
VAR _temInd = COUNTROWS( FILTER( _pendModelo, NOT ISBLANK( [Ind] ) ) ) + 0 > 0
// _semAbertura = nem a coluna do indicador nem a de abertura estao ligadas
VAR _semAbertura = _qCarga > 0 && _qComAbertura = 0 && NOT _temInd
// _soInd = so a coluna do indicador esta ligada: separa perdidas x dentro, mas
// nao consegue dizer quem vence hoje ou amanha (isso precisa da abertura)
VAR _soInd = _qCarga > 0 && _qComAbertura = 0 && _temInd
// so as ordens das ATs do filtro (vale mesmo sem relacionamento com a dAT)
VAR _pend0 =
    FILTER( _pendModelo, ( NOT ISBLANK( [Abertura] ) || NOT ISBLANK( [Ind] ) ) && CONTAINSROW( _atsCtx, [AT] ) )
VAR _pend1 =
    ADDCOLUMNS(
        _pend0,
        "CorN",
            VAR _a0 = [AT]
            RETURN COALESCE( MAXX( FILTER( _atsTodas, [AT] = _a0 ), [Cor] ), "" ),
        "AbN", IF( ISBLANK( [Abertura] ), BLANK(), INT( [Abertura] ) ),
        "Fora", IF( ISBLANK( [Ind] ), 0, IF( [Ind] = _valorDentro, 0, 1 ) ),
        "AgN", IF( ISBLANK( [Agenda] ), BLANK(), INT( [Agenda] ) )
    )
// A coluna Agenda so aparece quando informa alguma coisa: se todas as ordens
// estao agendadas para hoje (carga D-0 pura), ela seria uma coluna de "hoje".
VAR _qAgDiferente = COUNTROWS( FILTER( _pend1, NOT ISBLANK( [AgN] ) && [AgN] <> _hojeN ) ) + 0
VAR _qAgVazia = COUNTROWS( FILTER( _pend1, ISBLANK( [AgN] ) ) ) + 0
VAR _qAgCheia = COUNTROWS( FILTER( _pend1, NOT ISBLANK( [AgN] ) ) ) + 0
VAR _temAgenda = _qAgDiferente > 0 || ( _qAgVazia > 0 && _qAgCheia > 0 )
VAR _pend2 =
    ADDCOLUMNS(
        _pend1,
        // Idade = prazo ja consumido ate o fim de hoje; Prev = ate o fim de ontem.
        // Cada dia depois da criacao pesa 1 (util), 0,75 (sabado) ou 0,25 (domingo).
        "Idade",
            VAR _ab1 = [AbN]
            RETURN
                IF(
                    ISBLANK( _ab1 ),
                    BLANK(),
                    SUMX(
                        GENERATESERIES( MAX( _ab1 + 1, _hojeN - 30 ), _hojeN ),
                        VAR _w1 = WEEKDAY( [Value], 2 )
                        RETURN IF( _w1 <= 5, _pesoUtil, IF( _w1 = 6, _pesoSab, _pesoDom ) )
                    ) + 0
                ),
        "Prev",
            VAR _ab2 = [AbN]
            RETURN
                IF(
                    ISBLANK( _ab2 ),
                    BLANK(),
                    SUMX(
                        GENERATESERIES( MAX( _ab2 + 1, _hojeN - 30 ), _hojeN - 1 ),
                        VAR _w2 = WEEKDAY( [Value], 2 )
                        RETURN IF( _w2 <= 5, _pesoUtil, IF( _w2 = 6, _pesoSab, _pesoDom ) )
                    ) + 0
                ),
        // a data de criacao e a propria data agendada: nao ha agenda separada para comparar
        "Vence", BLANK(),
        "ClsC",
            VAR _c8 = [CorN]
