BSC Projeção HTML =

// ============================================================================
// BSC - PROJECAO E DIAGNOSTICO  (v1)   |   corpo da pagina "Projeção BSC"
// Colocar em um HTML Content ocupando o corpo da pagina. Fundo TRANSPARENTE.
// Tamanho de referencia: 987 x 576 (o mesmo do corpo do BSC).
//
// PERGUNTA QUE A PAGINA RESPONDE
//   "Se o indicador tivesse o atingimento esperado (a meta), qual seria a nota
//   do BSC?"  Ex.: Instalacao em ate D3 vale 10 pts e hoje entrega 5,0. Na
//   meta ele entrega os 10 -> a nota sobe 5,0.
//
// DOIS SELETORES NATIVOS COMANDAM A SIMULACAO (tabelas do arquivo 1)
//   'Seletor Cenário BSC'[Cenário]
//       "Atingimento real"             -> sem projecao: so o diagnostico
//       "Atingimento esperado (meta)"  -> os indicadores marcados contam com
//                                         o peso inteiro (padrao se nada for
//                                         marcado neste seletor)
//   'Seletor Projeção BSC'[Indicador Projeção]
//       quais indicadores entram na projecao. Desmarcar = TIRAR o indicador:
//       ele continua com o atingimento real. Nada marcado = todos.
//
// REGRAS
//   - Pontos atuais: a mesma medida [BSC Pontos Reais] e os mesmos filtros que
//     o corpo do BSC usa para a nota (Geral, cor ou AT). A "nota atual" daqui
//     e a nota do velocimetro do BSC.
//   - Ganho do indicador = peso x atingimento esperado - pontos atuais.
//     Nunca e negativo: quem ja passou da meta mantem o que tem.
//   - Nota projetada = nota atual + ganho dos indicadores que estao na projecao.
//   - Visao por AT: Indisponibilidade FTTH e Movel nao entram na nota da AT
//     (como no BSC), entao ficam fora da simulacao.
//   - Indicador sem resultado no filtro: hoje entrega 0 ponto; na projecao
//     entra com o peso inteiro (desmarque no seletor se nao quiser contar).
//
// PALETA: troque so a palavra em _tema ->  "GRAFITE" | "OCEANO" | "AURORA"
// (use o mesmo tema das outras medidas HTML).
//
// COMPATIBILIDADE COM O HTML CONTENT (nao executa JavaScript)
//   - so HTML + CSS + SVG embutido; sem <input>/<label>/<button>;
//   - atributo style so carrega variaveis CSS e fica em um <div class='v'>
//     neutro; valores globais (velocimetro, guias) vao direto no <style>;
//   - 100vh de proposito: o visual envolve o conteudo em um wrapper sem altura;
//   - nao usar aspas duplas dentro do CSS/HTML (quebram a string DAX).
// ============================================================================

// ------------------------------------------------------------------- PALETA
VAR _tema = "GRAFITE"

// d0/d1/d2 = fundos escuros | a1/a2 = cores de destaque | g1/g2 = brilhos
// pri/pri-b = destaque sobre fundo claro
VAR _tokens =
    SWITCH(
        _tema,
        "AURORA",
            "--d0:#140A2E;--d1:#22114F;--d2:#43208A;--a1:#F472B6;--a2:#A78BFA;--g1:rgba(244,114,182,.24);--g2:rgba(167,139,250,.30);--pri:#7C3AED;--pri-b:#F3EDFF",
        "GRAFITE",
            "--d0:#0C1016;--d1:#161C26;--d2:#2B3442;--a1:#7DD3FC;--a2:#CBD5E1;--g1:rgba(125,211,252,.14);--g2:rgba(148,163,184,.20);--pri:#334155;--pri-b:#EEF2F6",
        "--d0:#050E26;--d1:#0A1A42;--d2:#123A7C;--a1:#22D3EE;--a2:#60A5FA;--g1:rgba(34,211,238,.22);--g2:rgba(59,130,246,.30);--pri:#2563EB;--pri-b:#E9F0FF"
    )

// Escala do velocimetro: 130% = teto de pontuacao por indicador.
VAR _tetoBarra = 1.30

// ============================================================================
// SELECAO DE COR E AT  (igual ao corpo do BSC)
// ============================================================================
VAR _loc = "pt-BR"
VAR _corSelecionada =
    SELECTEDVALUE(
        'Seletor Cor'[COR_AREA],
        "GERAL"
    )
VAR _areaSelecionada =
    SELECTEDVALUE(
        dAT[AREA ]
    )
VAR _temAreaSelecionada =
    HASONEVALUE(
        dAT[AREA ]
    )
VAR _corEhFaixa =
    _corSelecionada = "OURO" || _corSelecionada = "BRONZE"
        || _corSelecionada = "PRATA 1" || _corSelecionada = "PRATA 2"

// cor da AT selecionada: a nota por AT so vale se a AT for da cor filtrada
VAR _corDaAT =
    IF(
        _temAreaSelecionada,
        CALCULATE(
            SELECTEDVALUE( dAT[COR_AREA ], "GERAL" ),
            REMOVEFILTERS( dAT[AREA ] ),
            REMOVEFILTERS( dAT[COR_AREA ] ),
            TREATAS( { _areaSelecionada }, dAT[AREA ] )
        )
    )
VAR _atValida =
    _temAreaSelecionada
        && ( ( NOT _corEhFaixa ) || ( _corDaAT = _corSelecionada ) )

// O que esta selecionado: AT > cor > Geral
VAR _modo = SWITCH( TRUE(), _atValida, "AT", _corEhFaixa, "COR", "GERAL" )

// ============================================================================
// SIMULACAO: o que os dois seletores nativos pedem
// ============================================================================
// Texto EXATO da opcao "sem projecao" da tabela 'Seletor Cenário BSC'.
VAR _nomeCenarioReal = "Atingimento real"
VAR _cenarioSel =
    SELECTEDVALUE(
        'Seletor Cenário BSC'[Cenário],
        "Atingimento esperado (meta)"
    )
VAR _simularPedido = _cenarioSel <> _nomeCenarioReal

// Atingimento esperado: 1 = 100% da meta. (Com 1.30 a pagina simula o teto.)
VAR _atingEsperado = 1

// Indicadores marcados no seletor (nada marcado = todos).
VAR _indMarcados =
    VALUES(
        'Seletor Projeção BSC'[Chave]
    )

// ============================================================================
// CALCULO - INDICADORES (mesma lista do corpo do BSC + sigla para os graficos)
// ============================================================================
VAR _indicadores =
    DATATABLE(
        "Indicador", STRING,
        "NomeTela", STRING,
        "Sigla", STRING,
        "Grupo", STRING,
        "Ordem", INTEGER,
        "AnaliticoAT", INTEGER,
        {
            { "1 AGENDA", "Efetividade 1ª Agenda", "AGE", "NEGÓCIO", 1, 1 },
            { "TMR", "Tempo de Reparo", "TMR", "NEGÓCIO", 2, 1 },
            { "CHEGADA NO SLOT", "Chegada no Slot", "SLT", "NEGÓCIO", 3, 1 },
            { "INSTALAÇÃO 3D", "Instalação em até D3", "D3", "NEGÓCIO", 4, 1 },
            { "PRODUTIVIDADE PU", "Produtividade PU", "PU", "NEGÓCIO", 5, 1 },
            { "ROTEIRIZAÇÃO AUTOMÁTICA", "Roteirização Automática", "ROT", "NEGÓCIO", 6, 1 },
            { "IRT", "IRT", "IRT", "NEGÓCIO", 7, 1 },
            { "EFICIÊNCIA DE CAMPO", "Eficiência de Campo", "EFC", "NEGÓCIO", 8, 1 },
            { "IFI 7 DIAS", "IFI FTTH 7 Dias", "IFI", "QUALIDADE", 9, 1 },
            { "IRR 7 DIAS", "Repetidas FTTH 7 Dias", "IRR", "QUALIDADE", 10, 1 },
            { "INDISPONIBILIDADE FTTH", "Indisponibilidade FTTH", "FTTH", "QUALIDADE", 11, 0 },
            { "INDISPONIBILIDADE MÓVEL", "Indisponibilidade Móvel", "MÓV", "QUALIDADE", 12, 0 }
        }
    )

VAR _totalPontosDisponiveis =
    SUMX(
        _indicadores,
        VAR _ind = [Indicador]
        RETURN
            CALCULATE(
                [BSC Pontos Disponíveis],
                REMOVEFILTERS( 'Seletor Cor' ),
                REMOVEFILTERS( 'Seletor Indicador BSC' ),
                TREATAS( { _ind }, 'Seletor Indicador BSC'[Indicador] )
            )
    )

// ============================================================================
// CALCULO - RESULTADO, META E PONTOS DE CADA INDICADOR NO FILTRO ATUAL
// __ResultadoBruto, __Meta, __MenorMelhor e __PtsDisp: copiados do corpo do BSC.
// __PtsBruto: [BSC Pontos Reais] com os filtros que o BSC usa para a nota
//             (Geral / cor / AT). Na visao por AT, so indicadores com
//             abertura por AT (AnaliticoAT = 1) tem pontos.
// ============================================================================
VAR _tabIndBase =
    ADDCOLUMNS(
        _indicadores,
        "__ResultadoBruto",
            VAR _ind = [Indicador]
            RETURN
                IF(
                    [AnaliticoAT] = 1,
                    CALCULATE(
                        [BSC Resultado Selecionado],
                        REMOVEFILTERS( 'Seletor Cor' ),
                        REMOVEFILTERS( 'Seletor Indicador BSC' ),
                        FILTER(
                            ALL( dAT[COR_AREA ] ),
                            _corSelecionada = "GERAL"
                                || dAT[COR_AREA ] = _corSelecionada
                        ),
                        TREATAS( { _ind }, 'Seletor Indicador BSC'[Indicador] )
                    ),
                    CALCULATE(
                        [BSC Resultado Selecionado],
                        REMOVEFILTERS( 'Seletor Cor' ),
                        REMOVEFILTERS( 'Seletor Indicador BSC' ),
                        REMOVEFILTERS( dAT[AREA ] ),
                        REMOVEFILTERS( dAT[COR_AREA ] ),
                        TREATAS( { _ind }, 'Seletor Indicador BSC'[Indicador] )
                    )
                ),
        "__Meta",
            VAR _ind = [Indicador]
            RETURN
                CALCULATE(
                    [BSC Meta Selecionada],
                    REMOVEFILTERS( 'Seletor Indicador BSC' ),
                    TREATAS( { _ind }, 'Seletor Indicador BSC'[Indicador] )
                ),
        "__MenorMelhor",
            VAR _ind = [Indicador]
            RETURN
                CALCULATE(
                    [BSC Menor Melhor],
                    REMOVEFILTERS( 'Seletor Indicador BSC' ),
                    TREATAS( { _ind }, 'Seletor Indicador BSC'[Indicador] )
                ),
        "__PtsDisp",
            VAR _ind = [Indicador]
            RETURN
                CALCULATE(
                    [BSC Pontos Disponíveis],
                    REMOVEFILTERS( 'Seletor Indicador BSC' ),
                    TREATAS( { _ind }, 'Seletor Indicador BSC'[Indicador] )
                ),
        "__PtsBruto",
            VAR _ind = [Indicador]
            RETURN
                SWITCH(
                    _modo,
                    "AT",
                        IF(
                            [AnaliticoAT] = 1,
                            CALCULATE(
                                [BSC Pontos Reais],
                                REMOVEFILTERS( 'Seletor Cor' ),
                                REMOVEFILTERS( 'Seletor Indicador BSC' ),
                                REMOVEFILTERS( dAT[AREA ] ),
                                TREATAS( { _areaSelecionada }, dAT[AREA ] ),
                                TREATAS( { _ind }, 'Seletor Indicador BSC'[Indicador] )
                            )
                        ),
                    "COR",
                        CALCULATE(
                            [BSC Pontos Reais],
                            REMOVEFILTERS( 'Seletor Cor' ),
                            REMOVEFILTERS( 'Seletor Indicador BSC' ),
                            REMOVEFILTERS( dAT[AREA ] ),
                            REMOVEFILTERS( dAT[COR_AREA ] ),
                            TREATAS( { _corSelecionada }, dAT[COR_AREA ] ),
                            TREATAS( { _ind }, 'Seletor Indicador BSC'[Indicador] )
                        ),
                    CALCULATE(
                        [BSC Pontos Reais],
                        REMOVEFILTERS( 'Seletor Cor' ),
                        REMOVEFILTERS( 'Seletor Indicador BSC' ),
                        REMOVEFILTERS( dAT[COR_AREA ] ),
                        REMOVEFILTERS( dAT[AREA ] ),
                        TREATAS( { _ind }, 'Seletor Indicador BSC'[Indicador] )
                    )
                )
    )

// -- guarda contra NaN / -nan(ind) vindo de divisao por zero na medida base
VAR _tabIndResultado =
    ADDCOLUMNS(
        _tabIndBase,
        "__Resultado",
            VAR _bruto = [__ResultadoBruto]
            VAR _invalido = SEARCH( "n", FORMAT( _bruto, "0.000000", "en-US" ), 1, 0 ) > 0
            RETURN
                IF( _invalido, BLANK(), _bruto )
    )

VAR _tabIndAting =
    ADDCOLUMNS(
        _tabIndResultado,
        "__Ating",
            VAR _res = [__Resultado]
            VAR _meta = [__Meta]
            RETURN
                IF(
                    ISBLANK( _res ) || ISBLANK( _meta ),
                    BLANK(),
                    IF(
                        [Indicador] = "IFI 7 DIAS" && _res = 0,
                        1.30,
                        IF(
                            [__MenorMelhor] = 1,
                            DIVIDE( _meta, _res ),
                            DIVIDE( _res, _meta )
                        )
                    )
                )
    )

// ============================================================================
// PROJECAO - PONTOS ATUAIS, OPORTUNIDADE E GANHO DE CADA INDICADOR
// ============================================================================
VAR _tabIndPts =
    ADDCOLUMNS(
        _tabIndAting,
        // 1 = o indicador compoe a nota deste filtro
        "__NaNota", IF( _modo = "AT" && [AnaliticoAT] = 0, 0, 1 ),
        // regra do BSC: IFI 7 dias com resultado 0% recebe o teto de 130%
        "__PtsReais",
            IF(
                [Indicador] = "IFI 7 DIAS" && ( NOT ISBLANK( [__Resultado] ) ) && [__Resultado] = 0,
                [__PtsDisp] * 1.30,
                [__PtsBruto]
            ),
        "__Faixa",
            SWITCH(
                TRUE(),
                ISBLANK( [__Ating] ), "neutral",
                [__Ating] >= 1, "green",
                [__Ating] >= 0.90, "yellow",
                [__Ating] >= 0.80, "orange",
                "red"
            ),
        // 1 = esta marcado no seletor de indicadores
        "__Marcado",
            VAR _ind = [Indicador]
            RETURN
                IF( ISEMPTY( FILTER( _indMarcados, 'Seletor Projeção BSC'[Chave] = _ind ) ), 0, 1 )
    )

// Oportunidade = pontos que faltam para o indicador entregar o esperado.
// Nao depende dos seletores (e o diagnostico).
VAR _tabIndPot =
    ADDCOLUMNS(
        _tabIndPts,
        "__Potencial",
            VAR _p = [__PtsDisp] * _atingEsperado - COALESCE( [__PtsReais], 0 )
            RETURN
                IF( [__NaNota] = 1 && _p >= 0.005, _p, 0 )
    )

// Nota atual = soma dos pontos dos indicadores que compoem a nota do filtro.
VAR _ptsAtual = SUMX( FILTER( _tabIndPot, [__NaNota] = 1 ), [__PtsReais] )
VAR _temNota = NOT ISBLANK( _ptsAtual )
// sem nenhum resultado no filtro nao ha o que projetar
VAR _simular = _simularPedido && _temNota

// Ganho = oportunidade que entrou na projecao (cenario esperado + marcado).
VAR _tabIndGanho =
    ADDCOLUMNS(
        _tabIndPot,
        "__Ganho", IF( _simular && [__Marcado] = 1, [__Potencial], 0 )
    )
VAR _ganhoSim = COALESCE( SUMX( _tabIndGanho, [__Ganho] ), 0 )
VAR _potencialTotal = COALESCE( SUMX( _tabIndGanho, [__Potencial] ), 0 )
VAR _ptsProj = COALESCE( _ptsAtual, 0 ) + _ganhoSim
// teto = nota se TODOS os indicadores entregassem o esperado
VAR _ptsTeto = COALESCE( _ptsAtual, 0 ) + _potencialTotal
VAR _temGanho = _ganhoSim >= 0.005

VAR _atingAtual = DIVIDE( _ptsAtual, _totalPontosDisponiveis )
VAR _atingProj = IF( _temNota, DIVIDE( _ptsProj, _totalPontosDisponiveis ) )
VAR _faixaAtual =
    SWITCH(
        TRUE(),
        ISBLANK( _atingAtual ), "neutral",
        _atingAtual >= 1, "green",
        _atingAtual >= 0.90, "yellow",
        _atingAtual >= 0.80, "orange",
        "red"
    )
VAR _faixaProj =
    SWITCH(
        TRUE(),
        ISBLANK( _atingProj ), "neutral",
        _atingProj >= 1, "green",
        _atingProj >= 0.90, "yellow",
        _atingProj >= 0.80, "orange",
        "red"
    )

// ---- escala da ponte: da nota atual (esquerda) ate a meta ou o teto ----
VAR _d0 = COALESCE( _ptsAtual, 0 )
VAR _d1 = MAX( _totalPontosDisponiveis, _ptsTeto )
VAR _dom = MAX( _d1 - _d0, _totalPontosDisponiveis * 0.10 )

// Estado do indicador na simulacao e chave de ordenacao da lista:
//   sim  = esta na projecao (ordem: maior ganho primeiro)
//   pot  = tem oportunidade, mas ficou com o atingimento real
//   ok   = ja entrega o esperado
//   fora = nao compoe a nota deste filtro
VAR _tabIndChave =
    ADDCOLUMNS(
        _tabIndGanho,
        "__Estado",
            SWITCH(
                TRUE(),
                [__NaNota] = 0, "fora",
                [__Ganho] > 0, "sim",
                [__Potencial] > 0, "pot",
                "ok"
            ),
        "__Chave",
            SWITCH(
                TRUE(),
                [__NaNota] = 0, 4000 + [Ordem],
                [__Ganho] > 0, 1000 - [__Ganho] + [Ordem] / 100000,
                [__Potencial] > 0, 2000 - [__Potencial] + [Ordem] / 100000,
                3000 + [Ordem]
            ),
        // posicao pelo tamanho da oportunidade (1 = maior), usada no caminho mais curto
        "__PosPot",
            VAR _p = [__Potencial]
            VAR _o = [Ordem]
            RETURN
                IF(
                    _p > 0,
                    1 + COUNTROWS(
                        FILTER(
                            _tabIndGanho,
                            [__Potencial] > _p
                                || ( [__Potencial] = _p && [Ordem] < _o )
                        )
                    )
                )
    )

VAR _tabInd =
    ADDCOLUMNS(
        _tabIndChave,
        // nota em que a barra do indicador comeca na ponte
        "__Ini",
            VAR _k = [__Chave]
            RETURN
                SWITCH(
                    [__Estado],
                    "sim",
                        _d0 + COALESCE(
                            SUMX( FILTER( _tabIndChave, [__Estado] = "sim" && [__Chave] < _k ), [__Ganho] ),
                            0
                        ),
                    "pot", _ptsProj
                ),
        // nota acumulada recuperando da maior oportunidade ate esta
        "__AcumPot",
            VAR _pp = [__PosPot]
            RETURN
                IF(
                    NOT ISBLANK( _pp ),
                    _d0 + SUMX(
                        FILTER( _tabIndChave, ( NOT ISBLANK( [__PosPot] ) ) && [__PosPot] <= _pp ),
                        [__Potencial]
                    )
                )
    )

VAR _qtdNaNota = COUNTROWS( FILTER( _tabInd, [__NaNota] = 1 ) ) + 0
VAR _qtdOport = COUNTROWS( FILTER( _tabInd, [__Potencial] > 0 ) ) + 0
VAR _qtdSim = COUNTROWS( FILTER( _tabInd, [__Estado] = "sim" ) ) + 0
VAR _qtdVerde = COUNTROWS( FILTER( _tabInd, [__Faixa] = "green" ) ) + 0
VAR _qtdAmarelo = COUNTROWS( FILTER( _tabInd, [__Faixa] = "yellow" ) ) + 0
VAR _qtdLaranja = COUNTROWS( FILTER( _tabInd, [__Faixa] = "orange" ) ) + 0
VAR _qtdVermelho = COUNTROWS( FILTER( _tabInd, [__Faixa] = "red" ) ) + 0
VAR _qtdSemDados = COUNTROWS( FILTER( _tabInd, [__Faixa] = "neutral" ) ) + 0

// ============================================================================
// APRESENTACAO - FAIXA DE DESTAQUE 1/3: NOTA ATUAL x NOTA PROJETADA
// ============================================================================
VAR _tituloSel =
    SWITCH( _modo, "AT", "AT " & _areaSelecionada, "COR", "BSC " & _corSelecionada, "BSC Geral" )
VAR _faltaProj = _totalPontosDisponiveis - _ptsProj

// ---- geometria do velocimetro (meio circulo), a mesma do corpo do BSC ----
// arco principal: raio 84 -> comprimento 263,9 | arco das zonas: raio 95 -> 298,5
VAR _arcoL = 263.9
VAR _arcoZ = 298.5
VAR _fracAtual = MIN( MAX( COALESCE( _atingAtual, 0 ), 0 ), _tetoBarra ) / _tetoBarra
VAR _fracProj = MIN( MAX( COALESCE( _atingProj, 0 ), 0 ), _tetoBarra ) / _tetoBarra
VAR _dashAtual = FORMAT( _fracAtual * _arcoL, "0.0", "en-US" )
VAR _dashProj = FORMAT( _fracProj * _arcoL, "0.0", "en-US" )
VAR _angMeta = FORMAT( 180 / _tetoBarra, "0.0", "en-US" )
VAR _z80 = 0.80 / _tetoBarra * _arcoZ
VAR _z90 = 0.90 / _tetoBarra * _arcoZ
VAR _z100 = 1 / _tetoBarra * _arcoZ
VAR _dashZ1 = FORMAT( _z80 - 2, "0.0", "en-US" ) & " 600"
VAR _dashZ2 = "0 " & FORMAT( _z80, "0.0", "en-US" ) & " " & FORMAT( _z90 - _z80 - 2, "0.0", "en-US" ) & " 600"
VAR _dashZ3 = "0 " & FORMAT( _z90, "0.0", "en-US" ) & " " & FORMAT( _z100 - _z90 - 2, "0.0", "en-US" ) & " 600"
VAR _dashZ4 = "0 " & FORMAT( _z100, "0.0", "en-US" ) & " " & FORMAT( _arcoZ - _z100, "0.0", "en-US" ) & " 600"

VAR _heroNota =
    "<div class='hz gz d-" & _faixaProj & "'>" &
        "<div class='gauge' title='" & _tituloSel & ": nota atual " & FORMAT( _ptsAtual, "0.00", _loc ) & " de " & FORMAT( _totalPontosDisponiveis, "0", _loc ) &
            " pts (" & FORMAT( _atingAtual, "0.0%", _loc ) & ")" &
            IF( _temGanho, " · projetada " & FORMAT( _ptsProj, "0.00", _loc ) & " pts (" & FORMAT( _atingProj, "0.0%", _loc ) & ")", "" ) &
            ". Arco cheio = nota atual; arco claro = nota projetada; traço branco = meta de 100%.'>" &
            "<svg viewBox='0 0 200 112' aria-hidden='true'>" &
                "<path class='g-z g-z1' d='M5 100A95 95 0 0 1 195 100'></path>" &
                "<path class='g-z g-z2' d='M5 100A95 95 0 0 1 195 100'></path>" &
                "<path class='g-z g-z3' d='M5 100A95 95 0 0 1 195 100'></path>" &
                "<path class='g-z g-z4' d='M5 100A95 95 0 0 1 195 100'></path>" &
                "<path class='g-t' d='M16 100A84 84 0 0 1 184 100'></path>" &
                IF( _temGanho, "<path class='g-p' d='M16 100A84 84 0 0 1 184 100'></path>", "" ) &
                IF( _fracAtual > 0, "<path class='g-f d-" & _faixaAtual & "' d='M16 100A84 84 0 0 1 184 100'></path>", "" ) &
                "<path class='g-m' d='M6 100h20'></path>" &
            "</svg>" &
            "<div class='g-v'><b>" & IF( _temNota, FORMAT( _ptsProj, "0.00", _loc ), "—" ) & "</b>" &
                "<small>" & IF( _temGanho, "projetada", "atual" ) & " · de " & FORMAT( _totalPontosDisponiveis, "0", _loc ) & " pts</small></div>" &
        "</div>" &
        "<div class='gt'>" &
            "<span class='gt-e'>" & _tituloSel & "</span>" &
            IF(
                NOT _temNota,
                "<span class='chip'>sem pontuação neste filtro</span>",
                "<span class='gt-s'>nota atual <b>" & FORMAT( _ptsAtual, "0.00", _loc ) & "</b> · " & FORMAT( _atingAtual, "0.0%", _loc ) & "</span>" &
                "<span class='chip'>" &
                    SWITCH(
                        TRUE(),
                        _temGanho, "▲ +" & FORMAT( _ganhoSim, "0.00", _loc ) & " pts na projeção",
                        _simular, "sem ganho nos indicadores marcados",
                        "cenário real · sem projeção"
