// ============================================================================
// VISUAL 2 de 2  -  BSC: FAIXAS + INDICADORES + RANKING DE ATS  (v3)
// Colocar em um HTML Content ocupando o restante da pagina.
// Fundo do visual: TRANSPARENTE.
// A barra de titulo/slicers e nativa do Power BI (fora deste visual).
//
// v3 - redesenho com foco em UX. As regras de calculo de pontos sao as mesmas
// da versao original; mudou a apresentacao:
//   - topo: Geral em destaque + 4 faixas ranqueadas, cada uma com delta vs.
//     Geral e medidor de atingimento (o traco no medidor = meta de 100%);
//   - tabela de indicadores orientada a acao: coluna "Falta" (pts ate a meta),
//     maior gap destacado e resumido no cabecalho;
//   - ranking de ATs com mapa de calor: 10 celulas por AT, uma por indicador,
//     mostrando de onde vem (ou nao vem) a pontuacao de cada AT;
//   - interacao feita so com CSS (o HTML Content nao executa JavaScript):
//       * mouse sobre uma faixa   -> o ranking destaca as ATs daquela faixa;
//       * mouse sobre uma AT      -> a faixa dela acende no topo;
//       * mouse sobre uma celula  -> mostra o valor e acende o mesmo indicador
//         em todas as ATs e na tabela (mouse na linha da tabela faz o inverso);
//       * AT selecionada no filtro fica fixa no topo/rodape da lista;
//   - escala de status validada para daltonismo, sempre com icone + valor.
// ============================================================================

    SUMX(

        VAR _ind = [Indicador]

VAR _atingGeral = DIVIDE( _pontosGeral, _totalPontosDisponiveis )

// ============================================================================
// RANKING DINÂMICO DOS QUATRO CARDS DE FAIXA
// ============================================================================
VAR _rankingBase =
    UNION(
        ROW("Cor", "OURO", "Pontos", COALESCE(_pontosOuro, 0)),
        ROW("Cor", "BRONZE", "Pontos", COALESCE(_pontosBronze, 0)),
        ROW("Cor", "PRATA 1", "Pontos", COALESCE(_pontosPrata1, 0)),
        ROW("Cor", "PRATA 2", "Pontos", COALESCE(_pontosPrata2, 0))
    )
VAR _rankingCards =
    ADDCOLUMNS(
        _rankingBase,
        "Posicao",
            VAR _corAtual = [Cor]
            VAR _pontosAtual = [Pontos]
            RETURN
                1 + COUNTROWS(
                    FILTER(
                        _rankingBase,
                        [Pontos] > _pontosAtual
                            || ([Pontos] = _pontosAtual && [Cor] < _corAtual)
                    )
                )
    )

// ============================================================================
// ESCALA DE STATUS (a mesma nos cards, na tabela e no ranking)
//   s1 >= 100% | s2 >= 90% | s3 >= 80% | s4 < 80% | s0 = sem dados
// ============================================================================

// ============================================================================
// KPI - GERAL (destaque) + 4 FAIXAS RANQUEADAS
// ============================================================================
VAR _stGeral =
    SWITCH(
        TRUE(),
        ISBLANK( _atingGeral ), "s0",
        _atingGeral >= 1, "s1",
        _atingGeral >= 0.90, "s2",
        _atingGeral >= 0.80, "s3",
        "s4"
    )
VAR _faltaGeral = _totalPontosDisponiveis - COALESCE( _pontosGeral, 0 )
VAR _cardGeral =
    "<div class='tile hero" & IF( _corSelecionada = "GERAL", " on", "" ) & "' style='--c:" & _corPrimaria & "'" &
        " title='Geral: " & FORMAT( _pontosGeral, "0.00", _loc ) & " de " & FORMAT( _totalPontosDisponiveis, "0", _loc ) &
        " pts disponíveis (" & FORMAT( _atingGeral, "0.0%", _loc ) & ")'>" &
        "<div class='t-top'><i class='dot'></i><span class='t-name'>GERAL</span>" &
            "<span class='t-n'>" & FORMAT( _qtdATGeral, "0", _loc ) & " ATs · " & FORMAT( COUNTROWS( _indicadores ), "0", _loc ) & " indicadores</span></div>" &
        "<div class='t-mid'><span class='t-val'>" & FORMAT( _pontosGeral, "0.00", _loc ) &
            "<small>de " & FORMAT( _totalPontosDisponiveis, "0", _loc ) & " pts</small></span>" &
            "<span class='st " & _stGeral & "'><i>" & IF( _atingGeral >= 1, "▲", "▼" ) & "</i>" & FORMAT( _atingGeral, "0.0%", _loc ) & "</span></div>" &
        "<div class='mt " & _stGeral & "'><u style='width:" & FORMAT( MIN( COALESCE( _atingGeral, 0 ), _tetoBarra ) / _tetoBarra * 100, "0.0", "en-US" ) & "%'></u></div>" &
        "<div class='t-foot'>" &
            IF(
                _faltaGeral > 0,
                "Faltam <b>" & FORMAT( _faltaGeral, "0.00", _loc ) & " pts</b> para 100%",
                "Meta batida · <b>+" & FORMAT( -_faltaGeral, "0.00", _loc ) & " pts</b> acima"
            ) &
        "</div>" &
    "</div>"

VAR _cardsFaixas =
    CONCATENATEX(
        _rankingCards,
        VAR _cor = [Cor]
        VAR _pts = [Pontos]
        VAR _pos = [Posicao]
        VAR _fx = SWITCH( _cor, "OURO", "ouro", "PRATA 1", "p1", "PRATA 2", "p2", "BRONZE", "bronze", "x" )
        VAR _hex = SWITCH( _cor, "OURO", _hexOuro, "PRATA 1", _hexPrata1, "PRATA 2", _hexPrata2, "BRONZE", _hexBronze, _corPrimaria )
        VAR _qtd = SWITCH( _cor, "OURO", _qtdATOuro, "BRONZE", _qtdATBronze, "PRATA 1", _qtdATPrata1, "PRATA 2", _qtdATPrata2 )
        VAR _ating = DIVIDE( _pts, _totalPontosDisponiveis )
        VAR _st =
            SWITCH(
                TRUE(),
                ISBLANK( _ating ), "s0",
                _ating >= 1, "s1",
                _ating >= 0.90, "s2",
                _ating >= 0.80, "s3",
                "s4"
            )
        VAR _delta = _pts - COALESCE( _pontosGeral, 0 )
        RETURN
            "<div class='tile fx-" & _fx & IF( _corSelecionada = _cor, " on", "" ) & IF( _pos = 1, " lead", "" ) & "' style='--c:" & _hex & "'" &
                " title='" & _cor & ": " & FORMAT( _pos, "0", _loc ) & "º lugar · " & FORMAT( _pts, "0.00", _loc ) & " pts (" &
                FORMAT( _ating, "0.0%", _loc ) & ") · " & FORMAT( _delta, "+0.00;-0.00;0.00", _loc ) & " pts vs. Geral. Passe o mouse para destacar as ATs da faixa.'>" &
                "<div class='t-top'><i class='dot'></i><span class='t-name'>" & _cor & "</span>" &
                    "<span class='t-n'>" & FORMAT( _qtd, "0", _loc ) & IF( _qtd = 1, " AT", " ATs" ) & "</span>" &
                    "<b class='t-rk'>" & FORMAT( _pos, "0", _loc ) & "º</b></div>" &
                "<div class='t-mid'><span class='t-val'>" & FORMAT( _pts, "0.00", _loc ) & "<small>pts</small></span>" &
                    "<span class='st " & _st & "'><i>" & IF( _ating >= 1, "▲", "▼" ) & "</i>" & FORMAT( _ating, "0.0%", _loc ) & "</span></div>" &
                "<div class='mt " & _st & "'><u style='width:" & FORMAT( MIN( COALESCE( _ating, 0 ), _tetoBarra ) / _tetoBarra * 100, "0.0", "en-US" ) & "%'></u></div>" &
                "<div class='t-foot'><span class='dl " & IF( _delta >= 0, "up", "down" ) & "'><i>" & IF( _delta >= 0, "▲", "▼" ) & "</i>" &
                    FORMAT( _delta, "+0.00;-0.00;0.00", _loc ) & "</span> vs. geral</div>" &
            "</div>",
        "",
        [Posicao], ASC
    )

// ============================================================================
// TABELA DOS 12 INDICADORES
// Primeiro uma tabela com os valores; depois o HTML. Subtotais, contadores
// da legenda e o "maior gap" saem da mesma fonte que as linhas exibidas.
// ============================================================================
VAR _tabIndBase =
    ADDCOLUMNS(
        _indicadores,
        "__ResultadoBruto",
            VAR _ind = [Indicador]
            RETURN
                IF(
                    [AnaliticoAT] = 1,
                    CALCULATE(
                        [BSC Resultado Selecionado],
                        REMOVEFILTERS( 'Seletor Cor' ),
                        REMOVEFILTERS( 'Seletor Indicador BSC' ),
                        FILTER(
                            ALL( dAT[COR_AREA ] ),
                            _corSelecionada = "GERAL"
                                || dAT[COR_AREA ] = _corSelecionada
                        ),
                        TREATAS( { _ind }, 'Seletor Indicador BSC'[Indicador] )
                    ),
                    CALCULATE(
                        [BSC Resultado Selecionado],
                        REMOVEFILTERS( 'Seletor Cor' ),
                        REMOVEFILTERS( 'Seletor Indicador BSC' ),
                        REMOVEFILTERS( dAT[AREA ] ),
                        REMOVEFILTERS( dAT[COR_AREA ] ),
                        TREATAS( { _ind }, 'Seletor Indicador BSC'[Indicador] )
                    )
                ),
        "__Meta",
            VAR _ind = [Indicador]
            RETURN
                CALCULATE(
                    [BSC Meta Selecionada],
                    REMOVEFILTERS( 'Seletor Indicador BSC' ),
                    TREATAS( { _ind }, 'Seletor Indicador BSC'[Indicador] )
                ),
        "__MenorMelhor",
            VAR _ind = [Indicador]
            RETURN
                CALCULATE(
                    [BSC Menor Melhor],
                    REMOVEFILTERS( 'Seletor Indicador BSC' ),
                    TREATAS( { _ind }, 'Seletor Indicador BSC'[Indicador] )
                ),
        "__PtsDisp",
            VAR _ind = [Indicador]
            RETURN
                CALCULATE(
                    [BSC Pontos Disponíveis],
                    REMOVEFILTERS( 'Seletor Indicador BSC' ),

    ADDCOLUMNS(
        _tabInd,
        "__Html",

            VAR _sRes = ROUND( _res * 3600, 0 )

                FORMAT( INT( MOD( _sRes, 3600 ) / 60 ), "00" ) & ":" &
                FORMAT( MOD( _sRes, 60 ), "00" )
            VAR _sMeta = ROUND( _meta * 3600, 0 )
            VAR _tmrMeta =
                FORMAT( INT( _sMeta / 3600 ), "00" ) & ":" &
                FORMAT( INT( MOD( _sMeta, 3600 ) / 60 ), "00" ) & ":" &
                FORMAT( MOD( _sMeta, 60 ), "00" )

                IF(
                    ISBLANK( _res ),

                )
            VAR _fmtMeta =
                IF(
                    ISBLANK( _meta ),
                    "—",
                    SWITCH(
                        TRUE(),
                        _ehTMR, _tmrMeta,
                        _ehQtd, FORMAT( _meta, "0.0", _loc ),
                        FORMAT( _meta, "0.0%", _loc )
                    )
                )

                IF(
                    [__MenorMelhor] = 1,
                    "<span class='tag' title='Quanto menor o resultado, melhor o desempenho'>↓ menor é melhor</span>",

                ) &
                IF(
                    [AnaliticoAT] = 0,

                )

                IF(
                    _semDados,

                            FORMAT( _ating, "0.0%", _loc ) &
                        "</span></div>"
                )
            VAR _celFalta =
                SWITCH(
                    TRUE(),
                    _semDados, "<span class='mu'>—</span>",
                    _ating >= 1, "<span class='ok'>✓ meta</span>",

            RETURN
                "<tr class='ir k" & _ordem & IF( _semDados, " nd", IF( _st = "s4", " al", "" ) ) & "'>" &

                        IF( _tags = "", "", "<div class='tags'>" & _tags & "</div>" ) & "</td>" &

                    "<td class='rl'>" & _fmtRes & "</td>" &
                    "<td>" & _celAting & "</td>" &
                    "<td>" & IF( _semDados, "<span class='mu'>—</span>", FORMAT( [__PtsReais], "0.0", _loc ) ) &
                        "<small>/" & FORMAT( [__PtsDisp], "0", _loc ) & "</small></td>" &

// ============================================================================
// RANKING DE ATS COM MAPA DE CALOR POR INDICADOR
// ============================================================================
// Indicadores que abrem por AT + pontos disponiveis de cada um
// (mesmo contexto usado no total de pontos disponiveis)
VAR _indRanking =
    ADDCOLUMNS(
        FILTER( _indicadores, [AnaliticoAT] = 1 ),
        "__Disp",
            VAR _ind = [Indicador]
            RETURN
                CALCULATE(
                    [BSC Pontos Disponíveis],
                    REMOVEFILTERS( 'Seletor Cor' ),
                    REMOVEFILTERS( 'Seletor Indicador BSC' ),
                    TREATAS( { _ind }, 'Seletor Indicador BSC'[Indicador] )
                )
    )
VAR _qtdCelNeg = COUNTROWS( FILTER( _indRanking, [Grupo] = "NEGÓCIO" ) ) + 0
VAR _qtdCelQua = COUNTROWS( FILTER( _indRanking, [Grupo] = "QUALIDADE" ) ) + 0
VAR _primeiraQua = MINX( FILTER( _indRanking, [Grupo] = "QUALIDADE" ), [Ordem] )
VAR _maxPtsArea = MAXX( _areasValidas, [__Pontos] )

VAR _linhasAreas =
    CONCATENATEX(
        _areasValidas,
        VAR _areaDaLinha = dAT[AREA ]
        VAR _pontosArea = [__Pontos]
        VAR _corArea = [__CorArea]
        VAR _ptAgenda = [__PontosAgenda]
        VAR _ptTMR = [__PontosTMR]
        VAR _ptSlot = [__PontosSlot]
        VAR _ptInstalacao = [__PontosInstalacao]
        VAR _ptProdutividade = [__PontosProdutividade]
        VAR _ptRoteirizacao = [__PontosRoteirizacao]
        VAR _ptIRT = [__PontosIRT]
        VAR _ptEficiencia = [__PontosEficienciaCampo]
        VAR _ptIFI7 = [__PontosIFI7]
        VAR _ptIRR7 = [__PontosIRR7]
        // posicao no ranking (mesmo criterio da ordenacao: pontos desc, AT asc)
        VAR _pos =
            1 + COUNTROWS(
                FILTER(
                    _areasValidas,
                    [__Pontos] > _pontosArea
                        || ( [__Pontos] = _pontosArea && dAT[AREA ] < _areaDaLinha )
                )
            )
        VAR _fx = SWITCH( _corArea, "OURO", "ouro", "PRATA 1", "p1", "PRATA 2", "p2", "BRONZE", "bronze", "x" )
        VAR _hex = SWITCH( _corArea, "OURO", _hexOuro, "PRATA 1", _hexPrata1, "PRATA 2", _hexPrata2, "BRONZE", _hexBronze, _corPrimaria )
        VAR _atingArea = DIVIDE( _pontosArea, _totalPontosDisponiveis )
        VAR _stArea =
            SWITCH(
                TRUE(),
                ISBLANK( _atingArea ), "s0",
                _atingArea >= 1, "s1",
                _atingArea >= 0.90, "s2",
                _atingArea >= 0.80, "s3",
                "s4"
            )
        VAR _selecionada = _temAreaSelecionada && _areaDaLinha = _areaSelecionada
        // uma celula por indicador: cor = atingimento da AT naquele indicador
        VAR _celulas =
            CONCATENATEX(
                _indRanking,
                VAR _o = [Ordem]
                VAR _disp = [__Disp]
                VAR _pi =
                    SWITCH(
                        [Indicador],
                        "1 AGENDA", _ptAgenda,
                        "TMR", _ptTMR,
                        "CHEGADA NO SLOT", _ptSlot,
                        "INSTALAÇÃO 3D", _ptInstalacao,
                        "PRODUTIVIDADE PU", _ptProdutividade,
                        "ROTEIRIZAÇÃO AUTOMÁTICA", _ptRoteirizacao,
                        "IRT", _ptIRT,
                        "EFICIÊNCIA DE CAMPO", _ptEficiencia,
                        "IFI 7 DIAS", _ptIFI7,
                        "IRR 7 DIAS", _ptIRR7
                    )
                VAR _ai = DIVIDE( _pi, _disp )
                VAR _si =
                    SWITCH(
                        TRUE(),
                        ISBLANK( _ai ), "s0",
                        _ai >= 1, "s1",
                        _ai >= 0.90, "s2",
                        _ai >= 0.80, "s3",
                        "s4"
                    )
                RETURN
                    "<i class='c k" & _o & " " & _si & IF( _o = _primeiraQua, " g2", "" ) & "'><b><em>" & _areaDaLinha & " · " & [NomeTela] & "</em>" &
                        IF(
                            ISBLANK( _ai ),
                            "sem dados",
                            FORMAT( _ai, "0%", _loc ) & " · " & FORMAT( _pi, "0.0", _loc ) & " de " & FORMAT( _disp, "0", _loc ) & " pts"
                        ) &
                    "</b></i>",
                "",
                [Ordem], ASC
            )
        RETURN
            "<div class='at fx-" & _fx & IF( _selecionada, " sel", "" ) & IF( _pos <= 3, " top", "" ) & "' style='--c:" & _hex & "'>" &
                "<span class='at-rk'>" & FORMAT( _pos, "0", _loc ) & "</span>" &
                "<div class='at-id'><b>" & _areaDaLinha & "</b><small>" & _corArea & "</small></div>" &
                "<div class='strip'>" & _celulas & "</div>" &
                "<div class='at-p'><b>" & FORMAT( _pontosArea, "0.00", _loc ) & "</b>" &
                    "<small class='" & _stArea & "'><i></i>" & FORMAT( _atingArea, "0.0%", _loc ) & "</small>" &
                    "<u style='width:" & FORMAT( DIVIDE( _pontosArea, _maxPtsArea ) * 100, "0.0", "en-US" ) & "%'></u></div>" &
            "</div>",
        "",
        [__Pontos], DESC,
        dAT[AREA ], ASC
    )

VAR _cabecalhoRanking =
    "<div class='at-h'><span>#</span><span>AT</span>" &
        "<span class='strip-h'>" &
            IF( _qtdCelNeg > 0, "<em class='neg' style='--n:" & _qtdCelNeg & "'>Negócio</em>", "" ) &
            IF( _qtdCelQua > 0, "<em class='qua' style='--n:" & _qtdCelQua & "'>" & IF( _qtdCelQua <= 3, "Qual.", "Qualidade" ) & "</em>", "" ) &
        "</span>" &
        "<span>Pontos</span></div>"

VAR _conteudoAreas =
    IF(
        _qtdAreasListadas = 0,
        "<div class='empty'><b>Nenhuma AT com pontuação</b>Ajuste os filtros de período, faixa ou AT.</div>",
        _cabecalhoRanking & _linhasAreas
    )
