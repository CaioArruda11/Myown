Pagina IRR HTML =
// ===========================================================================
// PAGINA IRR  (v3.1 - mesma linguagem visual das paginas BSC v3 e IFI v3.1)
// HTML Content ocupando a pagina. Fundo do visual: TRANSPARENTE.
// As consultas e medidas sao as mesmas da versao anterior; mudou a
// apresentacao:
//   - KPIs: IRR com meta, medidor e variacao vs. mes anterior; reparos
//     abertos e repetidos com minigrafico do periodo; Atingimento BSC em
//     medidor e Top 3 compacto;
//   - grafico: barras a partir do zero, linha de meta, cor por status,
//     rotulo so no pico e no ultimo ponto, outlier cortado com marcador e
//     variacao vs. periodo anterior no tooltip;
//   - BOTOES CLICAVEIS DENTRO DO VISUAL:
//       * Dia / Semana / Mes no proprio grafico (a tabela acompanha: em
//         Semana ela mostra as 3 ultimas semanas, nos demais os 3 meses -
//         mesma regra da versao anterior);
//       * Por AT / Por grupo na tabela de detalhamento.
//     O HTML Content nao executa JavaScript; os botoes usam <details> +
//     CSS :has() (as duas coisas passam no sanitizador do visual). Por isso
//     a medida entrega todas as visoes prontas e o clique so troca o que
//     aparece, na hora, sem recalcular. A visao inicial segue os seletores
//     nativos; qualquer mudanca de filtro redesenha o visual e ele volta
//     para essa visao inicial.
//   - interacao por mouse (so CSS):
//       * card do IRR ou legenda "acima da meta" -> destaca no grafico os
//         periodos acima da meta;
//       * AT do Top 3 -> destaca a linha dela na tabela.
// ===========================================================================
VAR _periodoIRR =
    SELECTEDVALUE('Seletor IRR'[Periodo IRR], "IRR 30D")

// Reaproveita somente o seletor existente de granularidade Dia/Semana/Mês.
VAR _granularidade =
    [IFI Granularidade Selecionada]

VAR _detalhamento =
    SELECTEDVALUE(
        'Seletor Detalhamento IRR'[Detalhamento],
        "Abertura por AT"
    )

VAR _irr =
    [IRR Percentual Selecionado]

VAR _reparosAbertos =
    [IRR Reparos Abertos Selecionado]

VAR _reparosRepetidos =
    [IRR Repetidos Selecionado]

VAR _atingimento =
    [BSC Atingimento]

VAR _irrTexto =
    IF(ISBLANK(_irr), "--", FORMAT(_irr, "0.0%"))

VAR _abertosTexto =
    IF(ISBLANK(_reparosAbertos), "--", FORMAT(_reparosAbertos, "#,##0"))

VAR _repetidosTexto =
    IF(ISBLANK(_reparosRepetidos), "--", FORMAT(_reparosRepetidos, "#,##0"))

VAR _atingimentoTexto =
    IF(ISBLANK(_atingimento), "--", FORMAT(_atingimento, "0.0%"))

// ---------------------------------------------------------------------------
// META DO IRR  (IRR e "menor e melhor")
// >>> CONFIRA ESTE BLOCO. A versao anterior da pagina nao tinha meta.
//   - _metaManual: preencha para fixar a meta (ex.: 0.06 = 6,0%). Vale para
//     qualquer periodo e tem prioridade.
//   - sem meta manual, nos periodos de 7 dias usa a meta do BSC do indicador
//     "IRR 7 DIAS" (Repetidas FTTH 7 Dias), a mesma da pagina BSC;
//   - nos periodos de 30 dias o modelo nao tem meta: sem _metaManual o visual
//     esconde a linha de meta e mostra as barras numa cor so.
// ---------------------------------------------------------------------------
VAR _metaManual = BLANK()
VAR _metaIRRBSC =
    CALCULATE(
        [BSC Meta Selecionada],
        REMOVEFILTERS('Seletor Indicador BSC'),
        TREATAS({"IRR 7 DIAS"}, 'Seletor Indicador BSC'[Indicador])
    )
VAR _metaIRR =
    COALESCE(
        _metaManual,
        IF(CONTAINSSTRING(_periodoIRR, "7D"), _metaIRRBSC)
    )
VAR _temMeta = NOT ISBLANK(_metaIRR) && _metaIRR > 0
VAR _metaTexto = IF(_temMeta, FORMAT(_metaIRR, "0.0%"), "--")
VAR _irrAcimaMeta = _temMeta && NOT ISBLANK(_irr) && _irr > _metaIRR

VAR _stAtingimento =
    SWITCH(
        TRUE(),
        ISBLANK(_atingimento), "s0",
        _atingimento >= 1, "s1",
        _atingimento >= 0.90, "s2",
        _atingimento >= 0.80, "s3",
        "s4"
    )
// Medidor do atingimento: escala ate 130% (teto do BSC); traco = 100%
VAR _tetoAtingimento = 1.30

// ---------------------------------------------------------------------------
// GRAFICO DE BARRAS DE VIDRO
// Usa a dCalendario para Dia, Semana e Mes.
// >>> ALTERADO: agora o eixo do grafico diario vai do dia 01 ate o ULTIMO dia
//     do mes (EOMONTH), e nao mais so ate a ultima data com dado. Dias futuros
//     do mes aparecem como barras "vazias" (classe .empty) mostrando "--".
// ---------------------------------------------------------------------------
VAR _dataMaxima =
    MAX(dCalendario[Data])
VAR _inicioMesGrafico =
    DATE(YEAR(_dataMaxima), MONTH(_dataMaxima), 1)
VAR _fimMesGrafico =
    EOMONTH(_dataMaxima, 0)                       // <<< ultimo dia do mes (dia 30/31/28...)
VAR _datasGrafico =
    FILTER(
        ALLSELECTED(dCalendario[Data]),
        dCalendario[Data] >= _inicioMesGrafico
            && dCalendario[Data] <= _fimMesGrafico   // <<< antes era _dataMaxima
    )
VAR _dadosDiaBase =
    ADDCOLUMNS(
        _datasGrafico,
        "__Rotulo", FORMAT(dCalendario[Data], "dd/MM"),
        "__RotuloCurto", FORMAT(dCalendario[Data], "dd"),   // <<< rotulo curto p/ caber 31 barras
        "__Ordem", RANKX(_datasGrafico, dCalendario[Data],, ASC, DENSE),
        "__IRR", CALCULATE([IRR Percentual Selecionado]),
        "__Anterior", CALCULATE([IRR Percentual Selecionado], DATEADD(dCalendario[Data], -1, DAY)),
        "__Altas", CALCULATE([IRR Repetidos Selecionado]),
        "__Reparos", CALCULATE([IRR Reparos Abertos Selecionado])
    )
VAR _semanasBase =
    CALCULATETABLE(
        SUMMARIZE(
            dCalendario,
            dCalendario[Semana_Range],
            "__DataInicio", MIN(dCalendario[Data])
        ),
        _datasGrafico
    )
VAR _dadosSemanaBase =
    ADDCOLUMNS(
        _semanasBase,
        "__Rotulo", dCalendario[Semana_Range],
        "__Ordem", RANKX(_semanasBase, [__DataInicio],, ASC, DENSE),
        "__IRR", CALCULATE([IRR Percentual Selecionado]),
        "__Anterior", CALCULATE([IRR Percentual Selecionado], DATEADD(dCalendario[Data], -7, DAY)),
        "__Altas", CALCULATE([IRR Repetidos Selecionado]),
        "__Reparos", CALCULATE([IRR Reparos Abertos Selecionado])
    )
VAR _mesesBase =
    CALCULATETABLE(
        SUMMARIZE(
            dCalendario,
            dCalendario[Mês e Ano],
            "__DataInicio", MIN(dCalendario[Data])
        ),
        DATESINPERIOD(dCalendario[Data], _dataMaxima, -6, MONTH)
    )
VAR _dadosMesBase =
    ADDCOLUMNS(
        _mesesBase,
        "__Rotulo", dCalendario[Mês e Ano],
        "__Ordem", RANKX(_mesesBase, [__DataInicio],, ASC, DENSE),
        "__IRR", CALCULATE([IRR Percentual Selecionado]),
        "__Anterior", CALCULATE([IRR Percentual Selecionado], DATEADD(dCalendario[Data], -1, MONTH)),
        "__Altas", CALCULATE([IRR Repetidos Selecionado]),
        "__Reparos", CALCULATE([IRR Reparos Abertos Selecionado])
    )

// ---------------------------------------------------------------------------
// GRAFICO (v3.1) - as 3 granularidades ficam prontas no HTML e os botoes
// Dia / Semana / Mes do proprio grafico trocam a visao sem recalcular.
// - escala comeca no zero (altura proporcional de verdade);
// - outlier: se um ponto for muito maior que o resto, a escala e cortada e a
//   barra recebe um marcador de corte + rotulo com o valor real;
// - rotulo fixo so no pico e no ultimo ponto; o resto fica no tooltip.
// ---------------------------------------------------------------------------
// visao inicial = selecao do seletor nativo
VAR _gran = IF(_granularidade IN {"Semana", "Mês"}, _granularidade, "Dia")

VAR _dadosGraficoUniao =
    UNION(
        SELECTCOLUMNS(
            _dadosDiaBase,
            "__Gran", "Dia",
            "__Rotulo", "" & [__Rotulo],
            "__RotuloEixo", "" & [__RotuloCurto],
            "__Ordem", [__Ordem],
            "__IRRBruto", [__IRR],
            "__Anterior", [__Anterior],
            "__Altas", [__Altas],
            "__Reparos", [__Reparos]
        ),
        SELECTCOLUMNS(
            _dadosSemanaBase,
            "__Gran", "Semana",
            "__Rotulo", "" & [__Rotulo],
            "__RotuloEixo", "" & [__Rotulo],
            "__Ordem", [__Ordem],
            "__IRRBruto", [__IRR],
            "__Anterior", [__Anterior],
            "__Altas", [__Altas],
            "__Reparos", [__Reparos]
        ),
        SELECTCOLUMNS(
            _dadosMesBase,
            "__Gran", "Mês",
            "__Rotulo", "" & [__Rotulo],
            "__RotuloEixo", "" & [__Rotulo],
            "__Ordem", [__Ordem],
            "__IRRBruto", [__IRR],
            "__Anterior", [__Anterior],
            "__Altas", [__Altas],
            "__Reparos", [__Reparos]
        )
    )

// -- guarda contra BLANK / NaN / infinito vindos de divisao por zero
VAR _dadosTodos =
    ADDCOLUMNS(
        _dadosGraficoUniao,
        "__V",
            VAR _b = [__IRRBruto]
            RETURN
                IF(
                    ISBLANK(_b) || SEARCH("n", FORMAT(_b, "0.000000", "en-US"), 1, 0) > 0,
                    BLANK(),
                    _b
                )
    )

VAR _granularidades =
    DATATABLE(
        "G", STRING,
        "OrdemG", INTEGER,
        "ClasseView", STRING,
        "Unidade", STRING,
        "RotuloAnterior", STRING,
        {
            { "Dia", 1, "v-dia", "dias", "vs. dia anterior" },
            { "Semana", 2, "v-sem", "semanas", "vs. semana anterior" },
            { "Mês", 3, "v-mes", "meses", "vs. mês anterior" }
        }
    )

VAR _viewsGrafico =
    CONCATENATEX(
        _granularidades,
        VAR _g = [G]
        VAR _classeView = [ClasseView]
        VAR _unidade = [Unidade]
        VAR _rotuloAnterior = [RotuloAnterior]
        VAR _dados = FILTER(_dadosTodos, [__Gran] = _g)
        VAR _qtdPontos = COUNTROWS(_dados) + 0
        VAR _comValor = FILTER(_dados, NOT ISBLANK([__V]))
        VAR _qtdComValor = COUNTROWS(_comValor) + 0
        VAR _maxV = MAXX(_comValor, [__V])
        VAR _p90 = IF(_qtdComValor >= 5, PERCENTILEX.INC(_comValor, [__V], 0.9), _maxV)

        // -- escala do eixo Y
        VAR _alvoEscala = MAX(IF(_temMeta, _metaIRR * 2, 0), COALESCE(_p90, 0) * 1.15)
        VAR _cortaOutlier = COALESCE(_maxV, 0) > _alvoEscala * 1.25
        VAR _topoBruto =
            IF(
                _cortaOutlier,
                _alvoEscala,
                MAX(COALESCE(_maxV, 0), IF(_temMeta, _metaIRR * 1.25, 0))
            )
        VAR _passo =
            SWITCH(
                TRUE(),
                _topoBruto <= 0.02, 0.005,
                _topoBruto <= 0.04, 0.01,
                _topoBruto <= 0.08, 0.02,
                _topoBruto <= 0.12, 0.03,
                _topoBruto <= 0.20, 0.05,
                _topoBruto <= 0.40, 0.10,
                0.20
            )
        VAR _nPassos = MAX(1, CEILING(ROUND(_topoBruto / _passo, 6), 1))
        VAR _yMax = _nPassos * _passo
        VAR _formatoEixo = IF(_passo < 0.01, "0.0%", "0%")
        VAR _gridlines =
            CONCATENATEX(
                GENERATESERIES(1, _nPassos, 1),
                VAR _valorLinha = [Value] * _passo
                RETURN
                    "<i class='gl' style='bottom:" & FORMAT(DIVIDE(_valorLinha, _yMax) * 100, "0.00", "en-US") & "%'><b>"
                        & FORMAT(_valorLinha, _formatoEixo) & "</b></i>",
                "",
                [Value], ASC
            )
        VAR _linhaMeta =
            IF(
                _temMeta,
                "<i class='ml' style='bottom:" & FORMAT(MIN(100, DIVIDE(_metaIRR, _yMax) * 100), "0.00", "en-US") & "%'>"
                    & "<b>meta " & _metaTexto & "</b></i>",
                ""
            )

        // -- pico, ultimo ponto e resumo
        VAR _pico = TOPN(1, _comValor, [__V], DESC, [__Ordem], ASC)
        VAR _picoOrdem = MAXX(_pico, [__Ordem])
        VAR _picoValor = MAXX(_pico, [__V])
        VAR _picoRotulo = MAXX(_pico, [__Rotulo])
        VAR _ultimo = TOPN(1, _comValor, [__Ordem], DESC)
        VAR _ultimoOrdem = MAXX(_ultimo, [__Ordem])
        VAR _ultimoValor = MAXX(_ultimo, [__V])
        VAR _ultimoRotulo = MAXX(_ultimo, [__Rotulo])
        VAR _qtdAcima = IF(_temMeta, COUNTROWS(FILTER(_comValor, [__V] > _metaIRR)) + 0, 0)
        VAR _resumo =
            IF(
                _qtdComValor = 0,
                "Sem dados no período selecionado",
                IF(_temMeta, "<b>" & _qtdAcima & " de " & _qtdComValor & "</b> " & _unidade & " acima da meta · ", "")
                    & "pico <b>" & FORMAT(_picoValor, "0.0%") & "</b> em " & _picoRotulo
                    & IF(
                        _ultimoOrdem <> _picoOrdem,
                        " · último <b>" & FORMAT(_ultimoValor, "0.0%") & "</b> em " & _ultimoRotulo,
                        ""
                    )
            )

        // -- barras
        VAR _barras =
            CONCATENATEX(
                _dados,
                VAR _v = [__V]
                VAR _ordem = [__Ordem]
                VAR _ant = [__Anterior]
                VAR _antValido =
                    IF(
                        ISBLANK(_ant) || _ant = 0 || SEARCH("n", FORMAT(_ant, "0.000000", "en-US"), 1, 0) > 0,
                        BLANK(),
                        _ant
                    )
                VAR _variacao =
                    IF(ISBLANK(_v) || ISBLANK(_antValido), BLANK(), DIVIDE(_v - _antValido, _antValido))
                VAR _vazio = ISBLANK(_v)
                VAR _acima = _temMeta && NOT _vazio && _v > _metaIRR
                VAR _cortada = NOT _vazio && _v > _yMax
                VAR _altura = IF(_vazio, 0, MIN(100, DIVIDE(_v, _yMax) * 100))
                VAR _rotulada = NOT _vazio && (_ordem = _picoOrdem || _ordem = _ultimoOrdem)
                VAR _borda =
                    SWITCH(
                        TRUE(),
                        _qtdPontos > 6 && _ordem <= 3, " tl",
                        _qtdPontos > 6 && _ordem > _qtdPontos - 3, " tr",
                        ""
                    )
                VAR _tooltip =
                    "<span class='tip'><b>" & [__Rotulo] & "</b>"
                        & "<span><i class='" & IF(_vazio, "nd", IF(_acima, "over", "ok")) & "'></i>IRR<strong>"
                            & IF(_vazio, "--", FORMAT(_v, "0.0%")) & "</strong></span>"
                        & IF(
                            ISBLANK(_variacao),
                            "",
                            "<span>" & _rotuloAnterior & "<strong class='" & IF(_variacao > 0, "pior", "melhor") & "'>"
                                & FORMAT(_variacao, "+0%;-0%;0%") & "</strong></span>"
                        )
                        & "<span>Reparos repetidos<strong>" & FORMAT(COALESCE([__Altas], 0), "#,##0") & "</strong></span>"
                        & "<span>Reparos abertos<strong>" & FORMAT(COALESCE([__Reparos], 0), "#,##0") & "</strong></span>"
                    & "</span>"
                RETURN
                    "<div class='col" & IF(_vazio, " vazio", IF(_acima, " over", "")) & IF(_cortada, " cut", "")
                        & IF(_ordem = _ultimoOrdem, " last", "") & _borda & "'>"
                        & "<div class='pc'><i class='bar' style='height:" & FORMAT(_altura, "0.0", "en-US") & "%'>"
                            & IF(_rotulada, "<em>" & FORMAT(_v, "0.0%") & "</em>", "")
                        & "</i>" & _tooltip & "</div>"
                        & "<span class='xl'>" & [__RotuloEixo] & "</span>"
                    & "</div>",
                "",
                [__Ordem], ASC
            )
        RETURN
            "<div class='view " & _classeView & IF(_g = _gran, " ini", "") & "'>"
                & "<p class='vs'>" & _resumo & "</p>"
                & "<div class='plot'>"
                    & "<div class='grid'>" & _gridlines & _linhaMeta & "</div>"
                    & "<div class='cols" & IF(_qtdPontos > 16, " many", "") & "'>" & _barras & "</div>"
                & "</div>"
            & "</div>",
        "",
        [OrdemG], ASC
    )

// -- minigraficos dos cards de reparos (granularidade do seletor nativo)
// (na base do grafico, __Altas = reparos repetidos e __Reparos = reparos abertos)
VAR _dadosGrafico = FILTER(_dadosTodos, [__Gran] = _gran)
VAR _ultimoOrdemSpark = MAXX(FILTER(_dadosGrafico, NOT ISBLANK([__V])), [__Ordem])
VAR _maxRepetidos = MAXX(_dadosGrafico, COALESCE([__Altas], 0))
VAR _maxReparos = MAXX(_dadosGrafico, COALESCE([__Reparos], 0))
VAR _sparkRepetidos =
    CONCATENATEX(
        _dadosGrafico,
        "<i" & IF([__Ordem] = _ultimoOrdemSpark, " class='cur'", "") & " style='height:"
            & FORMAT(MAX(0, DIVIDE(COALESCE([__Altas], 0), _maxRepetidos) * 100), "0", "en-US") & "%'></i>",
        "",
        [__Ordem], ASC
    )
VAR _sparkReparos =
    CONCATENATEX(
        _dadosGrafico,
        "<i" & IF([__Ordem] = _ultimoOrdemSpark, " class='cur'", "") & " style='height:"
            & FORMAT(MAX(0, DIVIDE(COALESCE([__Reparos], 0), _maxReparos) * 100), "0", "en-US") & "%'></i>",
        "",
        [__Ordem], ASC
    )

// ---------------------------------------------------------------------------
// RANKING DAS ATS - IRR menor e melhor
// ---------------------------------------------------------------------------
VAR _areasRanking =
    FILTER(ALLSELECTED(dAT[AREA ]), NOT ISBLANK(dAT[AREA ]))
VAR _rankingBase =
    FILTER(
        ADDCOLUMNS(
            _areasRanking,
            "__IRR", CALCULATE([IRR Percentual Selecionado])
        ),
        ISNUMBER([__IRR]) && NOT ISBLANK([__IRR])
    )
VAR _top3 =
    TOPN(3, _rankingBase, [__IRR], ASC, dAT[AREA ], ASC)
VAR _comPosicao =
    ADDCOLUMNS(
        _top3,
        "__Posicao",
            VAR _valorAtual = [__IRR]
            VAR _areaAtual = dAT[AREA ]
            RETURN
                1 + COUNTROWS(
                    FILTER(
                        _top3,
                        [__IRR] < _valorAtual
                            || ([__IRR] = _valorAtual && dAT[AREA ] < _areaAtual)
                    )
                )
    )
VAR _at1 = MAXX(FILTER(_comPosicao, [__Posicao] = 1), dAT[AREA ])
VAR _at2 = MAXX(FILTER(_comPosicao, [__Posicao] = 2), dAT[AREA ])
VAR _at3 = MAXX(FILTER(_comPosicao, [__Posicao] = 3), dAT[AREA ])
VAR _irr1 = MAXX(FILTER(_comPosicao, [__Posicao] = 1), [__IRR])
VAR _irr2 = MAXX(FILTER(_comPosicao, [__Posicao] = 2), [__IRR])
VAR _irr3 = MAXX(FILTER(_comPosicao, [__Posicao] = 3), [__IRR])

// ---------------------------------------------------------------------------
// TOP 3 (substitui o podio) - classes "p-XX" ligam cada AT a sua linha na
// tabela (classe "r-XX") para o destaque ao passar o mouse.
// ---------------------------------------------------------------------------
VAR _cls1 = SUBSTITUTE(SUBSTITUTE(SUBSTITUTE(TRIM(COALESCE(_at1, "")), " ", "_"), ".", "_"), "/", "_")
VAR _cls2 = SUBSTITUTE(SUBSTITUTE(SUBSTITUTE(TRIM(COALESCE(_at2, "")), " ", "_"), ".", "_"), "/", "_")
VAR _cls3 = SUBSTITUTE(SUBSTITUTE(SUBSTITUTE(TRIM(COALESCE(_at3, "")), " ", "_"), ".", "_"), "/", "_")
VAR _linhaTop =
    "<div class='pd p-" & _cls1 & "'><span class='md m1'>1</span><b>" & COALESCE(_at1, "--") & "</b><span class='pv'>"
        & IF(ISBLANK(_irr1), "--", FORMAT(_irr1, "0.0%")) & "</span></div>"
    & "<div class='pd p-" & _cls2 & "'><span class='md m2'>2</span><b>" & COALESCE(_at2, "--") & "</b><span class='pv'>"
        & IF(ISBLANK(_irr2), "--", FORMAT(_irr2, "0.0%")) & "</span></div>"
    & "<div class='pd p-" & _cls3 & "'><span class='md m3'>3</span><b>" & COALESCE(_at3, "--") & "</b><span class='pv'>"
        & IF(ISBLANK(_irr3), "--", FORMAT(_irr3, "0.0%")) & "</span></div>"
VAR _cssTop3 =
    IF(_cls1 = "", "", ".dash:has(.p-" & _cls1 & ":hover) tr.r-" & _cls1 & " td,")
    & IF(_cls2 = "", "", ".dash:has(.p-" & _cls2 & ":hover) tr.r-" & _cls2 & " td,")
    & IF(_cls3 = "", "", ".dash:has(.p-" & _cls3 & ":hover) tr.r-" & _cls3 & " td,")
    & ".dash-nada{background-color:#EEECFD!important}"

// ---------------------------------------------------------------------------
// CONTEXTO DINAMICO DA TABELA: 3 MESES OU 3 SEMANAS
// Quando o seletor estiver em Semana, a tabela acompanha as ultimas 3 semanas.
// ---------------------------------------------------------------------------
VAR _dataReferenciaTabela = MAX(dCalendario[Data])
VAR _ehTabelaSemanal = _granularidade = "Semana"

VAR _fimMesAtual = EOMONTH(_dataReferenciaTabela, 0)
VAR _inicioMesAtual = EOMONTH(_fimMesAtual, -1) + 1
VAR _inicioMesM1 = EOMONTH(_fimMesAtual, -2) + 1
VAR _fimMesM1 = EOMONTH(_fimMesAtual, -1)
VAR _inicioMesM2 = EOMONTH(_fimMesAtual, -3) + 1
VAR _fimMesM2 = EOMONTH(_fimMesAtual, -2)

VAR _inicioSemanaAtual = _dataReferenciaTabela - WEEKDAY(_dataReferenciaTabela, 2) + 1
VAR _fimSemanaAtual = _inicioSemanaAtual + 6
VAR _inicioSemanaS1 = _inicioSemanaAtual - 7
VAR _fimSemanaS1 = _fimSemanaAtual - 7
VAR _inicioSemanaS2 = _inicioSemanaAtual - 14
VAR _fimSemanaS2 = _fimSemanaAtual - 14

VAR _inicioP0 = IF(_ehTabelaSemanal, _inicioSemanaAtual, _inicioMesAtual)
VAR _fimP0 = IF(_ehTabelaSemanal, _fimSemanaAtual, _fimMesAtual)
VAR _inicioP1 = IF(_ehTabelaSemanal, _inicioSemanaS1, _inicioMesM1)
VAR _fimP1 = IF(_ehTabelaSemanal, _fimSemanaS1, _fimMesM1)
VAR _inicioP2 = IF(_ehTabelaSemanal, _inicioSemanaS2, _inicioMesM2)
VAR _fimP2 = IF(_ehTabelaSemanal, _fimSemanaS2, _fimMesM2)

VAR _rotuloP0 =
    IF(
        _ehTabelaSemanal,
        FORMAT(_inicioSemanaAtual, "dd/MM") & " a " & FORMAT(_fimSemanaAtual, "dd/MM"),
        FORMAT(_fimMesAtual, "MMM/yy")
    )
VAR _rotuloP1 =
    IF(
        _ehTabelaSemanal,
        FORMAT(_inicioSemanaS1, "dd/MM") & " a " & FORMAT(_fimSemanaS1, "dd/MM"),
